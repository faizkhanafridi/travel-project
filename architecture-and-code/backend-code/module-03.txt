namespace App\Models\Agent;

use App\Enums\Agent\AgentStatus;
use App\Models\User;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Illuminate\Database\Eloquent\Relations\HasMany;
use Illuminate\Database\Eloquent\Relations\HasOne;
use Illuminate\Support\Str;

class Agent extends Model
{
    protected $fillable = [
        'uuid',
        'user_id',
        'status',
        'current_attempt',
        'submitted_at',
        'approved_at',
        'rejected_at',
    ];

    protected $casts = [
        'status' => AgentStatus::class,
        'submitted_at' => 'datetime',
        'approved_at' => 'datetime',
        'rejected_at' => 'datetime',
        'current_attempt' => 'integer',
    ];

    protected static function booted(): void
    {
        static::creating(function (self $agent) {
            $agent->uuid ??= (string) Str::uuid();
        });
    }

    public function user(): BelongsTo
    {
        return $this->belongsTo(User::class);
    }

    public function profile(): HasOne
    {
        return $this->hasOne(AgentProfile::class);
    }

    public function documents(): HasMany
    {
        return $this->hasMany(AgentDocument::class);
    }

    public function businessHours(): HasMany
    {
        return $this->hasMany(AgentBusinessHour::class)->orderBy('day_of_week');
    }

    public function attempts(): HasMany
    {
        return $this->hasMany(AgentApplicationAttempt::class)->orderBy('attempt_number');
    }

    public function events(): HasMany
    {
        return $this->hasMany(AgentApplicationEvent::class)->latest();
    }

    public function isEditable(): bool
    {
        return in_array($this->status, [
            AgentStatus::DRAFT,
            AgentStatus::REJECTED,
        ], true);
    }

    public function isApproved(): bool
    {
        return $this->status === AgentStatus::APPROVED;
    }

    public function isLocked(): bool
    {
        return in_array($this->status, [
            AgentStatus::PENDING_REVIEW,
            AgentStatus::UNDER_REVIEW,
            AgentStatus::SUSPENDED,
        ], true);
    }
}<?php

namespace App\Repositories\Eloquent\Agent;

use App\Enums\Agent\AgentStatus;
use App\Models\Agent\Agent;
use App\Models\Agent\AgentApplicationAttempt;
use App\Models\Agent\AgentApplicationEvent;

use App\Models\Agent\AgentProfile;
use App\Models\User;
use App\Repositories\Contracts\Agent\AgentRepositoryInterface;

class AgentRepository implements AgentRepositoryInterface
{
    public function findByUser(User $user): ?Agent
    {
        return Agent::where('user_id', $user->id)->first();
    }

    public function findByUuid(string $uuid): ?Agent
    {
        return Agent::where('uuid', $uuid)->first();
    }

    public function createForUser(User $user): Agent
    {
        return Agent::create([
            'user_id' => $user->id,
            'status' => AgentStatus::DRAFT,
            'current_attempt' => 1,
        ]);
    }

    public function update(Agent $agent, array $data): Agent
    {
        $agent->update($data);

        return $agent->refresh();
    }

    public function saveProfile(Agent $agent, array $agency = [], array $business = []): void
    {
        $profile = $agent->profile ?? new AgentProfile(['agent_id' => $agent->id]);
        $profile->fill(array_merge($agency, $business));
        $profile->agent_id = $agent->id;
        $profile->save();
    }

    public function replaceBusinessHours(Agent $agent, array $hours): void
    {
        $agent->businessHours()->delete();

        foreach ($hours as $hour) {
            $agent->businessHours()->create([
                'day_of_week' => $hour['day_of_week'],
                'is_closed' => $hour['is_closed'] ?? false,
                'open_time' => $hour['is_closed'] ?? false ? null : ($hour['open_time'] ?? null),
                'close_time' => $hour['is_closed'] ?? false ? null : ($hour['close_time'] ?? null),
            ]);
        }
    }

    public function addDocument(Agent $agent, array $data): void
    {
        $agent->documents()->create($data);
    }

    public function deleteDocument(string $uuid, Agent $agent): bool
    {
        $doc = $agent->documents()->where('uuid', $uuid)->first();

        if (! $doc) {
            return false;
        }

        return (bool) $doc->delete();
    }

    public function markApprovedDocuments(Agent $agent): void
    {
        $agent->documents()->update(['is_approved' => true]);
    }

    public function createAttempt(Agent $agent, array $data): AgentApplicationAttempt
    {
        return AgentApplicationAttempt::create(array_merge($data, [
            'agent_id' => $agent->id,
        ]));
    }

    public function logEvent(Agent $agent, array $data): void
    {
        AgentApplicationEvent::create(array_merge($data, ['agent_id' => $agent->id]));
    }

    public function paginateForAdmin(array $filters, int $perPage = 20)
    {
        return Agent::query()
            ->with(['user', 'profile', 'attempts' => fn ($q) => $q->latest('attempt_number')->limit(1)])
            ->when($filters['status'] ?? null, fn ($q, $s) => $q->where('status', $s))
            ->when($filters['search'] ?? null, function ($q, $s) {
                $q->where(function ($inner) use ($s) {
                    $inner->whereHas('user', fn ($u) => $u->where('email', 'like', "%{$s}%"))
                        ->orWhereHas('profile', fn ($p) => $p->where('agency_name', 'like', "%{$s}%")
                            ->orWhere('agency_owner_name', 'like', "%{$s}%"));
                });
            })
            ->latest()
            ->paginate($perPage);
    }
}<?php

namespace App\Services\Agent;

use App\Enums\Agent\AgentStatus;
use App\Enums\Agent\ApplicationEventType;
use App\Enums\Agent\DocumentType;
use App\Models\Agent\Agent;
use App\Models\Agent\AgentDocument;
use App\Models\User;
use App\Repositories\Contracts\Agent\AgentRepositoryInterface;
use Illuminate\Http\UploadedFile;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Storage;
use Illuminate\Validation\ValidationException;

class AgentApplicationService
{
    public function __construct(
        private AgentRepositoryInterface $agentRepo,
    ) {}

    /**
     * Get or create a draft agent row for the given user. Blocked if the user
     * already has an active agent record that is not rejected.
     */
    public function startOrResume(User $user): Agent
    {
        // Must have a user profile (Module 02) before applying.
        if (! $user->profile) {
            throw ValidationException::withMessages([
                'profile' => ['Complete your user profile before applying to become an agent.'],
            ]);
        }

        $existing = $this->agentRepo->findByUser($user);

        if ($existing) {
            if ($existing->status === AgentStatus::REJECTED) {
                return $existing; // can resume/edit and resubmit
            }

            if ($existing->status === AgentStatus::DRAFT) {
                return $existing;
            }

            // Pending / Under Review / Approved / Suspended — cannot start again.
            throw ValidationException::withMessages([
                'agent' => ['You already have an active agent application or account.'],
            ]);
        }

        return DB::transaction(fn () => $this->agentRepo->createForUser($user));
    }

    public function saveAgencyInformation(Agent $agent, array $data): Agent
    {
        $this->assertEditable($agent);

        $this->agentRepo->saveProfile($agent, [
            'agency_name' => $data['agency_name'],
            'agency_owner_name' => $data['agency_owner_name'],
            'phone' => $data['phone'] ?? null,
            'country' => $data['country'],
            'city' => $data['city'],
            'address' => $data['address'],
            'agency_information_completed' => true,
        ]);

        return $agent->refresh();
    }

    public function saveBusinessInformation(Agent $agent, array $data): Agent
    {
        $this->assertEditable($agent);
        $this->assertStepCompleted($agent, 'agency_information_completed');

        $this->agentRepo->saveProfile($agent, [], [
            'business_description' => $data['business_description'],
            'years_in_business' => $data['years_in_business'],
            'business_type' => $data['business_type'],
            'registration_information' => $data['registration_information'] ?? null,
            'website' => $data['website'] ?? null,
            'social_media' => $data['social_media'] ?? [],
            'languages' => $data['languages'] ?? [],
            'specializations' => $data['specializations'] ?? [],
            'business_information_completed' => true,
        ]);

        return $agent->refresh();
    }

    public function uploadDocument(Agent $agent, string $type, UploadedFile $file): AgentDocument
    {
        $this->assertEditable($agent);
        $this->assertStepCompleted($agent, 'business_information_completed');

        if (! in_array($type, array_map(fn ($t) => $t->value, DocumentType::cases()), true)) {
            throw ValidationException::withMessages(['type' => ['Invalid document type.']]);
        }

        // The unique types can only have one file at a time.
        if ($type !== DocumentType::OTHER->value) {
            $agent->documents()->where('type', $type)->delete();
        }

        // Store on private disk with a random filename; never trust client name.
        $disk = config('agent.document_disk');
        $path = config('agent.document_path');
        $storedPath = $file->store($path, $disk);

        $doc = null;
        DB::transaction(function () use ($agent, $type, $file, $storedPath, &$doc) {
            $this->agentRepo->addDocument($agent, [
                'type' => $type,
                'original_name' => $file->getClientOriginalName(),
                'storage_path' => $storedPath,
                'mime' => $file->getMimeType(),
                'size' => $file->getSize(),
            ]);
            $doc = $agent->documents()->where('storage_path', $storedPath)->first();
        });

        return $doc;
    }

    public function deleteDocument(Agent $agent, string $uuid): void
    {
        $this->assertEditable($agent);

        $doc = $agent->documents()->where('uuid', $uuid)->first();

        if (! $doc) {
            throw ValidationException::withMessages(['document' => ['Document not found.']]);
        }

        DB::transaction(function () use ($doc) {
            Storage::disk(config('agent.document_disk'))->delete($doc->storage_path);
            $doc->delete();
        });
    }

    public function saveBusinessHours(Agent $agent, array $hours): Agent
    {
        $this->assertEditable($agent);
        $this->assertStepCompleted($agent, 'business_information_completed');
        $this->assertRequiredDocumentsUploaded($agent);

        DB::transaction(function () use ($agent, $hours) {
            $this->agentRepo->replaceBusinessHours($agent, $hours);
        });

        return $agent->refresh();
    }

    /**
     * Review step guard: everything must be saved and required documents exist.
     */
    public function assertReadyForReview(Agent $agent): void
    {
        $this->assertEditable($agent);

        $profile = $agent->profile;
        if (! $profile || ! $profile->agency_information_completed || ! $profile->business_information_completed) {
            throw ValidationException::withMessages([
                'review' => ['Complete all previous steps before reviewing.'],
            ]);
        }

        $this->assertRequiredDocumentsUploaded($agent);

        if ($agent->businessHours()->count() < 7) {
            throw ValidationException::withMessages([
                'review' => ['Configure business hours for all 7 days before reviewing.'],
            ]);
        }
    }

    /**
     * Submit or resubmit. Creates a new attempt row and moves status to pending_review.
     */
    public function submit(Agent $agent, User $user): Agent
    {
        $this->assertReadyForReview($agent);

        return DB::transaction(function () use ($agent, $user) {
            $nextAttempt = $agent->attempts()->max('attempt_number') + 1;

            $this->agentRepo->createAttempt($agent, [
                'attempt_number' => $nextAttempt,
                'status' => AgentStatus::PENDING_REVIEW->value,
                'submitted_at' => now(),
            ]);

            $this->agentRepo->update($agent, [
                'status' => AgentStatus::PENDING_REVIEW,
                'current_attempt' => $nextAttempt,
                'submitted_at' => now(),
                'rejected_at' => null,
            ]);

            $this->agentRepo->logEvent($agent, [
                'event' => $nextAttempt === 1
                    ? ApplicationEventType::SUBMITTED->value
                    : ApplicationEventType::RESUBMITTED->value,
                'actor_id' => $user->id,
                'meta' => ['attempt' => $nextAttempt],
            ]);

            return $agent->refresh();
        });
    }

    private function assertEditable(Agent $agent): void
    {
        if (! $agent->isEditable()) {
            throw ValidationException::withMessages([
                'agent' => ['This application cannot be edited in its current status.'],
            ]);
        }
    }

    private function assertStepCompleted(Agent $agent, string $field): void
    {
        if (! $agent->profile || ! $agent->profile->{$field}) {
            throw ValidationException::withMessages([
                'step' => ['Previous step must be saved before continuing.'],
            ]);
        }
    }

    private function assertRequiredDocumentsUploaded(Agent $agent): void
    {
        foreach (DocumentType::required() as $required) {
            if (! $agent->documents()->where('type', $required->value)->exists()) {
                throw ValidationException::withMessages([

namespace App\Services\Agent;

use App\Enums\Agent\DocumentType;
use App\Models\Agent\Agent;
use App\Models\Agent\AgentDocument;
use Illuminate\Http\UploadedFile;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Storage;
use Illuminate\Validation\ValidationException;

class AgentDocumentService
{
    public function upload(Agent $agent, string $type, UploadedFile $file): AgentDocument
    {
        $this->assertEditable($agent);
        $this->assertBusinessCompleted($agent);

        if (! in_array($type, array_map(fn ($t) => $t->value, DocumentType::cases()), true)) {
            throw ValidationException::withMessages(['type' => ['Invalid document type.']]);
        }

        $disk = config('agent.document_disk');
        $path = config('agent.document_path');

        // Replace existing file for the same fixed type (business_registration / license / identity).
        // "other" type can have many files.
        if ($type !== DocumentType::OTHER->value) {
            foreach ($agent->documents()->where('type', $type)->get() as $doc) {
                Storage::disk($disk)->delete($doc->storage_path);
                $doc->delete();
            }
        }

        // Store on private disk with a random filename; never trust client name.
        $storedPath = $file->store($path, $disk);

        return DB::transaction(function () use ($agent, $type, $file, $storedPath) {
            $agent->documents()->create([
                'type' => $type,
                'original_name' => $file->getClientOriginalName(),
                'storage_path' => $storedPath,
                'mime' => $file->getMimeType(),
                'size' => $file->getSize(),
            ]);

            return $agent->documents()->latest('id')->first();
        });
    }

    public function delete(Agent $agent, string $uuid): void
    {
        $this->assertEditable($agent);

        $doc = $agent->documents()->where('uuid', $uuid)->first();

        if (! $doc) {
            throw ValidationException::withMessages(['document' => ['Document not found.']]);
        }

        DB::transaction(function () use ($doc) {
            Storage::disk(config('agent.document_disk'))->delete($doc->storage_path);
            $doc->delete();
        });
    }

    /**
     * Stream a private document. Caller must have already authorized
     * (owner or admin). Path never exposed publicly.
     */
    public function download(Agent $agent, string $uuid)
    {
        $doc = $agent->documents()->where('uuid', $uuid)->first();

        if (! $doc) {
            throw ValidationException::withMessages(['document' => ['Document not found.']]);
        }

        $disk = Storage::disk(config('agent.document_disk'));

        if (! $disk->exists($doc->storage_path)) {
            throw ValidationException::withMessages(['document' => ['File is missing on storage.']]);
        }

        return $disk->download($doc->storage_path, $doc->original_name);
    }

    private function assertEditable(Agent $agent): void
    {
        if (! $agent->isEditable()) {
            throw ValidationException::withMessages([
                'agent' => ['Documents cannot be edited in the current application status.'],
            ]);
        }
    }

    private function assertBusinessCompleted(Agent $agent): void
    {
        if (! $agent->profile || ! $agent->profile->business_information_completed) {
            throw ValidationException::withMessages([
                'step' => ['Complete business information before uploading documents.'],
            ]);
        }
    }

namespace App\Services\Agent;

use App\Enums\Agent\AgentStatus;
use App\Enums\Agent\ApplicationEventType;
use App\Enums\Auth\RoleName;
use App\Models\Agent\Agent;
use App\Models\User;
use App\Repositories\Contracts\Agent\AgentRepositoryInterface;
use Illuminate\Support\Facades\DB;
use Illuminate\Validation\ValidationException;

class AgentReviewService
{
    public function __construct(
        private AgentRepositoryInterface $agentRepo,
    ) {}

    public function markUnderReview(Agent $agent, User $admin): Agent
    {
        if (! in_array($agent->status, [AgentStatus::PENDING_REVIEW], true)) {
            throw ValidationException::withMessages(['agent' => ['Only pending applications can be marked under review.']]);
        }

        return DB::transaction(function () use ($agent, $admin) {
            $this->agentRepo->update($agent, ['status' => AgentStatus::UNDER_REVIEW]);
            $this->agentRepo->logEvent($agent, [
                'event' => ApplicationEventType::UNDER_REVIEW->value,
                'actor_id' => $admin->id,
            ]);

            return $agent->refresh();
        });
    }

    public function approve(Agent $agent, User $admin): Agent
    {
        if (! in_array($agent->status, [AgentStatus::PENDING_REVIEW, AgentStatus::UNDER_REVIEW], true)) {
            throw ValidationException::withMessages(['agent' => ['This application cannot be approved in its current status.']]);
        }

        return DB::transaction(function () use ($agent, $admin) {
            $this->agentRepo->update($agent, [
                'status' => AgentStatus::APPROVED,
                'approved_at' => now(),
            ]);

            // Assign the agent role to the user. This is the ONLY place that grants agent role.
            $agent->user->assignRole(RoleName::AGENT->value);

            // Freeze documents as approved (view-only).
            $this->agentRepo->markApprovedDocuments($agent);

            // Update the latest attempt.
            $latest = $agent->attempts()->latest('attempt_number')->first();
            if ($latest) {
                $latest->update([
                    'status' => AgentStatus::APPROVED->value,
                    'reviewed_at' => now(),
                    'reviewed_by' => $admin->id,
                ]);
            }

            $this->agentRepo->logEvent($agent, [
                'event' => ApplicationEventType::APPROVED->value,
                'actor_id' => $admin->id,
            ]);

            return $agent->refresh();
        });
    }

    public function reject(Agent $agent, User $admin, string $reason): Agent
    {
        if (! in_array($agent->status, [AgentStatus::PENDING_REVIEW, AgentStatus::UNDER_REVIEW], true)) {
            throw ValidationException::withMessages(['agent' => ['This application cannot be rejected in its current status.']]);
        }

        if (trim($reason) === '') {
            throw ValidationException::withMessages(['rejection_reason' => ['A rejection reason is required.']]);
        }

        return DB::transaction(function () use ($agent, $admin, $reason) {
            $this->agentRepo->update($agent, [
                'status' => AgentStatus::REJECTED,
                'rejected_at' => now(),
            ]);

            $latest = $agent->attempts()->latest('attempt_number')->first();
            if ($latest) {
                $latest->update([
                    'status' => AgentStatus::REJECTED->value,
                    'reviewed_at' => now(),
                    'reviewed_by' => $admin->id,
                    'rejection_reason' => $reason,
                ]);
            }

            $this->agentRepo->logEvent($agent, [
                'event' => ApplicationEventType::REJECTED->value,
                'actor_id' => $admin->id,
                'note' => $reason,
            ]);

            return $agent->refresh();
        });
    }

    public function suspend(Agent $agent, User $admin, string $reason): Agent
    {
        if ($agent->status !== AgentStatus::APPROVED) {
            throw ValidationException::withMessages(['agent' => ['Only approved agents can be suspended.']]);
        }

        return DB::transaction(function () use ($agent, $admin, $reason) {
            $this->agentRepo->update($agent, ['status' => AgentStatus::SUSPENDED]);
            $agent->user->removeRole(RoleName::AGENT->value);
            $this->agentRepo->logEvent($agent, [
                'event' => ApplicationEventType::SUSPENDED->value,
                'actor_id' => $admin->id,
                'note' => $reason,
            ]);

            return $agent->refresh();
        });
    }

    public function unsuspend(Agent $agent, User $admin): Agent
    {
        if ($agent->status !== AgentStatus::SUSPENDED) {
            throw ValidationException::withMessages([
                'agent' => ['Only suspended agents can be unsuspended.'],
            ]);
        }

        return DB::transaction(function () use ($agent, $admin) {
            $this->agentRepo->update($agent, ['status' => AgentStatus::APPROVED]);

            // Re-grant the agent role — this is the ONLY other place (besides approve)
            // that grants the role, and only after review has confirmed identity.
            $agent->user->assignRole(RoleName::AGENT->value);

            $this->agentRepo->logEvent($agent, [
                'event' => ApplicationEventType::UNSUSPENDED->value,
                'actor_id' => $admin->id,
            ]);

            return $agent->refresh();
        });
    }

namespace App\Http\Controllers\Agent;

use App\Http\Controllers\Controller;
use App\Http\Requests\Agent\SaveAgencyInformationRequest;
use App\Http\Requests\Agent\SaveBusinessHoursRequest;
use App\Http\Requests\Agent\SaveBusinessInformationRequest;
use App\Http\Resources\Agent\AgentResource;
use App\Services\Agent\AgentApplicationService;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Validation\ValidationException;

class AgentApplicationController extends Controller
{
    public function __construct(
        private AgentApplicationService $applicationService,
    ) {}

    // GET /api/agent/application — returns current application or null.
    public function show(Request $request): JsonResponse
    {
        $user = $request->user();
        $agent = $user->agent; // relationship added below

        return response()->json([
            'success' => true,
            'message' => $agent ? 'Application retrieved.' : 'No agent application yet.',
            'data' => $agent ? new AgentResource(
                $agent->load(['profile', 'documents', 'businessHours', 'attempts'])
            ) : null,
        ]);
    }

    // POST /api/agent/application/start — creates or resumes draft.
    public function start(Request $request): JsonResponse
    {
        $agent = $this->applicationService->startOrResume($request->user());

        return response()->json([
            'success' => true,
            'message' => 'Application started.',
            'data' => new AgentResource(
                $agent->load(['profile', 'documents', 'businessHours', 'attempts'])
            ),
        ]);
    }

    // PUT /api/agent/application/agency
    public function saveAgency(SaveAgencyInformationRequest $request): JsonResponse
    {
        $agent = $this->requireEditable($request);
        $agent = $this->applicationService->saveAgencyInfo($agent, $request->validated());

        return response()->json([
            'success' => true,
            'message' => 'Agency information saved.',
            'data' => new AgentResource(
                $agent->load(['profile', 'documents', 'businessHours', 'attempts'])
            ),
        ]);
    }

    // PUT /api/agent/application/business
    public function saveBusiness(SaveBusinessInformationRequest $request): JsonResponse
    {
        $agent = $this->requireEditable($request);
        $agent = $this->applicationService->saveBusinessInfo($agent, $request->validated());

        return response()->json([
            'success' => true,
            'message' => 'Business information saved.',
            'data' => new AgentResource(
                $agent->load(['profile', 'documents', 'businessHours', 'attempts'])
            ),
        ]);
    }

    // PUT /api/agent/application/business-hours
    public function saveHours(SaveBusinessHoursRequest $request): JsonResponse
    {
        $agent = $this->requireEditable($request);
        $agent = $this->applicationService->saveBusinessHours(
            $agent,
            $request->validated()['hours'],
        );

        return response()->json([
            'success' => true,
            'message' => 'Business hours saved.',
            'data' => new AgentResource(
                $agent->load(['profile', 'documents', 'businessHours', 'attempts'])
            ),
        ]);
    }

    // GET /api/agent/application/review — shows review payload only if ready.
    public function review(Request $request): JsonResponse
    {
        $user = $request->user();
        $agent = $user->agent;

        if (! $agent) {
            throw ValidationException::withMessages(['agent' => ['Start an application first.']]);
        }

        $this->applicationService->assertReadyForReview($agent);

        return response()->json([
            'success' => true,
            'message' => 'Ready for review.',
            'data' => new AgentResource(
                $agent->load(['profile', 'documents', 'businessHours', 'attempts'])
            ),
        ]);
    }

    // POST /api/agent/application/submit
    public function submit(Request $request): JsonResponse
    {
        $user = $request->user();
        $agent = $user->agent;

        if (! $agent) {
            throw ValidationException::withMessages(['agent' => ['Start an application first.']]);
        }

        $agent = $this->applicationService->submit($agent, $user);

        return response()->json([
            'success' => true,
            'message' => 'Application submitted. It is now pending review.',
            'data' => new AgentResource(
                $agent->load(['profile', 'documents', 'businessHours', 'attempts'])
            ),
        ]);
    }

    private function requireEditable(Request $request)
    {
        $agent = $request->user()->agent;

        if (! $agent) {
            throw ValidationException::withMessages(['agent' => ['Start an application first.']]);
        }

        return $agent;
    }

namespace App\Http\Controllers\Agent;

use App\Http\Controllers\Controller;
use App\Http\Requests\Agent\UploadDocumentRequest;
use App\Http\Resources\Agent\AgentDocumentResource;
use App\Services\Agent\AgentDocumentService;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Validation\ValidationException;

class AgentDocumentController extends Controller
{
    public function __construct(
        private AgentDocumentService $documentService,
    ) {}

    // POST /api/agent/documents
    public function store(UploadDocumentRequest $request): JsonResponse
    {
        $agent = $request->user()->agent;

        if (! $agent) {
            throw ValidationException::withMessages(['agent' => ['Start an application first.']]);
        }

        $doc = $this->documentService->upload(
            $agent,
            $request->string('type')->toString(),
            $request->file('file'),
        );

        return response()->json([
            'success' => true,
            'message' => 'Document uploaded.',
            'data' => new AgentDocumentResource($doc),
        ], 201);
    }

    // DELETE /api/agent/documents/{uuid}
    public function destroy(Request $request, string $uuid): JsonResponse
    {
        $agent = $request->user()->agent;

        if (! $agent) {
            throw ValidationException::withMessages(['agent' => ['No application found.']]);
        }

        $this->documentService->delete($agent, $uuid);

        return response()->json([
            'success' => true,
            'message' => 'Document deleted.',
            'data' => null,
        ]);
    }

    // GET /api/agent/documents/{uuid}/download
    public function download(Request $request, string $uuid)
    {
        $agent = $request->user()->agent;

        if (! $agent) {
            throw ValidationException::withMessages(['agent' => ['No application found.']]);
        }

        return $this->documentService->download($agent, $uuid);
    }
