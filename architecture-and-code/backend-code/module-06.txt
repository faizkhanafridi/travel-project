namespace App\Repositories\Eloquent\Messaging;

use App\Models\Agent\Agent;
use App\Models\Auth\User;
use App\Models\Messaging\Conversation;
use App\Repositories\Contracts\Messaging\ConversationRepositoryInterface;
use Illuminate\Contracts\Pagination\LengthAwarePaginator;
use Illuminate\Support\Str;

class ConversationRepository implements ConversationRepositoryInterface
{
    public function findByUuid(string $uuid): ?Conversation
    {
        return Conversation::with(['customer', 'agentRecord.user', 'agentRecord.profile', 'travelService', 'lastMessage'])
            ->where('uuid', $uuid)
            ->first();
    }

    public function findForUser(string $uuid, User $user): ?Conversation
    {
        return Conversation::with(['customer', 'agentRecord.user', 'agentRecord.profile', 'travelService', 'lastMessage'])
            ->where('uuid', $uuid)
            ->where(function ($q) use ($user) {
                $q->where('customer_id', $user->id)
                  ->orWhereHas('agentRecord', fn ($a) => $a->where('user_id', $user->id));
            })
            ->first();
    }

    public function findBetweenCustomerAndAgent(int $customerId, int $agentId): ?Conversation
    {
        return Conversation::where('customer_id', $customerId)
            ->where('agent_id', $agentId)
            ->first();
    }

    public function create(array $data): Conversation
    {
        $data['uuid'] = $data['uuid'] ?? (string) Str::uuid();
        return Conversation::create($data);
    }

    public function paginateForUser(User $user, array $filters): LengthAwarePaginator
    {
        $query = Conversation::with(['customer', 'agentRecord.user', 'agentRecord.profile', 'lastMessage'])
            ->where(function ($q) use ($user) {
                $q->where('customer_id', $user->id)
                  ->orWhereHas('agentRecord', fn ($a) => $a->where('user_id', $user->id));
            })
            ->orderByDesc('last_message_at')
            ->orderByDesc('id');

        if (! empty($filters['search'])) {
            $search = $filters['search'];
            $query->where(function ($q) use ($search, $user) {
                $q->whereHas('customer', fn ($c) => $c->where('name', 'like', "%{$search}%"))
                  ->orWhereHas('agentRecord.user', fn ($a) => $a->where('name', 'like', "%{$search}%"))
                  ->orWhereHas('agentRecord.profile', fn ($p) => $p->where('agency_name', 'like', "%{$search}%"));
            });
        }

        return $query->paginate((int) ($filters['per_page'] ?? 20));
    }

    public function touchLastMessage(Conversation $conversation): void
    {
        $conversation->forceFill(['last_message_at' => now()])->save();
    }

namespace App\Repositories\Eloquent\Messaging;

use App\Models\Messaging\Conversation;
use App\Models\Messaging\Message;
use App\Repositories\Contracts\Messaging\MessageRepositoryInterface;
use Illuminate\Contracts\Pagination\LengthAwarePaginator;
use Illuminate\Support\Str;

class MessageRepository implements MessageRepositoryInterface
{
    public function create(array $data): Message
    {
        $data['uuid'] = $data['uuid'] ?? (string) Str::uuid();
        return Message::create($data);
    }

    public function paginateForConversation(Conversation $conversation, array $filters): LengthAwarePaginator
    {
        $query = Message::with('sender')
            ->where('conversation_id', $conversation->id)
            ->orderByDesc('created_at')
            ->orderByDesc('id');

        return $query->paginate((int) ($filters['per_page'] ?? 30));
    }

    public function markConversationRead(Conversation $conversation, int $userId): void
    {
        Message::where('conversation_id', $conversation->id)
            ->where('sender_id', '!=', $userId)
            ->whereNull('read_at')
            ->update(['read_at' => now()]);

        $conversation->participants()
            ->where('user_id', $userId)
            ->update(['last_read_at' => now()]);
    }

    public function unreadCount(Conversation $conversation, int $userId): int
    {
        return Message::where('conversation_id', $conversation->id)
            ->where('sender_id', '!=', $userId)
            ->whereNull('read_at')
            ->count();
    }

namespace App\Services\Messaging;

use App\Enums\Messaging\MessageType;
use App\Events\Messaging\MessageSent;
use App\Models\Agent\Agent;
use App\Models\Auth\User;
use App\Models\Messaging\Conversation;
use App\Models\Messaging\Message;
use App\Models\TravelService\TravelService;
use App\Repositories\Contracts\Messaging\ConversationRepositoryInterface;
use App\Repositories\Contracts\Messaging\MessageRepositoryInterface;
use Illuminate\Contracts\Pagination\LengthAwarePaginator;
use Illuminate\Support\Facades\DB;
use Symfony\Component\HttpKernel\Exception\AccessDeniedHttpException;
use Symfony\Component\HttpKernel\Exception\UnprocessableEntityHttpException;

class ConversationService
{
    public function __construct(
        private ConversationRepositoryInterface $conversations,
        private MessageRepositoryInterface $messages,
    ) {}

    /**
     * Open (or reuse) a conversation between the logged-in customer and a target agent.
     * Returns the conversation and whether it was newly created.
     *
     * @return array{conversation: Conversation, created: bool}
     */
    public function openConversation(User $customer, string $agentUuid, ?string $serviceUuid): array
    {
        $agent = Agent::with(['user', 'profile'])->where('uuid', $agentUuid)->first();
        if (! $agent) {
            throw new UnprocessableEntityHttpException('Agent not found.');
        }
        if ($agent->user_id === $customer->id) {
            throw new UnprocessableEntityHttpException('You cannot open a conversation with yourself.');
        }

        $service = null;
        if ($serviceUuid) {
            $service = TravelService::where('uuid', $serviceUuid)->first();
            if (! $service || $service->agent_id !== $agent->id) {
                throw new UnprocessableEntityHttpException('Service does not belong to this agent.');
            }
        }

        $existing = $this->conversations->findBetweenCustomerAndAgent($customer->id, $agent->id);
        if ($existing) {
            if ($service && $existing->travel_service_id !== $service->id) {
                $existing->forceFill(['travel_service_id' => $service->id])->save();
            }
            return ['conversation' => $this->conversations->findForUser($existing->uuid, $customer), 'created' => false];
        }

        $conversation = DB::transaction(function () use ($customer, $agent, $service) {
            $conv = $this->conversations->create([
                'type' => 'customer_agent',
                'customer_id' => $customer->id,
                'agent_id' => $agent->id,
                'travel_service_id' => $service?->id,
                'last_message_at' => null,
            ]);

            $conv->participants()->createMany([
                ['user_id' => $customer->id, 'last_read_at' => now()],
                ['user_id' => $agent->user_id, 'last_read_at' => null],
            ]);

            return $conv;
        });

        return ['conversation' => $this->conversations->findForUser($conversation->uuid, $customer), 'created' => true];
    }

    public function list(User $user, array $filters): LengthAwarePaginator
    {
        return $this->conversations->paginateForUser($user, $filters);
    }

    public function show(User $user, string $uuid): Conversation
    {
        $conversation = $this->conversations->findForUser($uuid, $user);
        if (! $conversation) {
            throw new AccessDeniedHttpException('Conversation not found or not yours.');
        }
        return $conversation;
    }

    public function listMessages(User $user, string $uuid, array $filters): LengthAwarePaginator
    {
        $conversation = $this->show($user, $uuid);

        // opening the message list marks as read
        $this->messages->markConversationRead($conversation, $user->id);

        return $this->messages->paginateForConversation($conversation, $filters);
    }

    public function sendMessage(User $user, string $uuid, string $body, array $meta = []): Message
    {
        $conversation = $this->show($user, $uuid);

        if (trim($body) === '') {
            throw new UnprocessableEntityHttpException('Message body is required.');
        }
        if (mb_strlen($body) > 4000) {
            throw new UnprocessableEntityHttpException('Message is too long.');
        }

        $message = DB::transaction(function () use ($conversation, $user, $body, $meta) {
            $message = $this->messages->create([
                'conversation_id' => $conversation->id,
                'sender_id' => $user->id,
                'type' => MessageType::TEXT->value,
                'body' => trim($body),
                'meta' => $meta ?: null,
            ]);

            $this->conversations->touchLastMessage($conversation);

            // sender's own read pointer advances
            $conversation->participants()
                ->where('user_id', $user->id)
                ->update(['last_read_at' => now()]);

            return $message->load(['sender', 'conversation']);
        });

        broadcast(new MessageSent($message))->toOthers();

        return $message;
    }

    public function markRead(User $user, string $uuid): void
    {
        $conversation = $this->show($user, $uuid);
        $this->messages->markConversationRead($conversation, $user->id);
    }

    public function unreadCount(User $user, string $uuid): int
    {
        $conversation = $this->show($user, $uuid);
        return $this->messages->unreadCount($conversation, $user->id);
    }

namespace App\Http\Controllers\Messaging;

use App\Http\Controllers\Controller;
use App\Http\Requests\Messaging\SendMessageRequest;
use App\Http\Requests\Messaging\StartConversationRequest;
use App\Http\Resources\Messaging\ConversationResource;
use App\Http\Resources\Messaging\MessageResource;
use App\Repositories\Contracts\Messaging\MessageRepositoryInterface;
use App\Services\Messaging\ConversationService;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;

class ConversationController extends Controller
{
    public function __construct(
        private ConversationService $service,
        private MessageRepositoryInterface $messages,
    ) {}

    public function index(Request $request): JsonResponse
    {
        $paginator = $this->service->list($request->user(), $request->only(['search', 'per_page']));

        $paginator->through(function ($conversation) use ($request) {
            $conversation->unread_count = $this->messages->unreadCount($conversation, $request->user()->id);
            return $conversation;
        });

        return ConversationResource::collection($paginator)->response();
    }

    public function start(StartConversationRequest $request): JsonResponse
    {
        $result = $this->service->openConversation(
            $request->user(),
            (string) $request->string('agent_uuid'),
            $request->input('service_uuid') ? (string) $request->input('service_uuid') : null,
        );

        $conversation = $result['conversation'];
        $conversation->unread_count = $this->messages->unreadCount($conversation, $request->user()->id);

        return (new ConversationResource($conversation))
            ->additional(['created' => $result['created']])
            ->response()
            ->setStatusCode($result['created'] ? 201 : 200);
    }

    public function show(Request $request, string $uuid): JsonResponse
    {
        $conversation = $this->service->show($request->user(), $uuid);
        $conversation->unread_count = $this->messages->unreadCount($conversation, $request->user()->id);

        return (new ConversationResource($conversation))->response();
    }

    public function messages(Request $request, string $uuid): JsonResponse
    {
        $paginator = $this->service->listMessages(
            $request->user(),
            $uuid,
            $request->only(['per_page']),
        );

        return MessageResource::collection($paginator)->response();
    }

    public function send(SendMessageRequest $request, string $uuid): JsonResponse
    {
        $message = $this->service->sendMessage(
            $request->user(),
            $uuid,
            (string) $request->string('body'),
            (array) ($request->input('meta') ?? []),
        );

            'unread' => $this->service->unreadCount($request->user(), $uuid),
        ]);
