class PendingRegistration extends Model
{
    protected $fillable = [
        'uuid',
        'email',
        'password',
        'verification_code',
        'attempts',
        'expires_at',
        'last_sent_at',
        'ip_address',
    ];

    protected $hidden = [
        'password',
        'verification_code',
    ];

    protected $casts = [
        'expires_at' => 'datetime',
        'last_sent_at' => 'datetime',
        'attempts' => 'integer',
    ];

    protected static function booted(): void
    {
        static::creating(function (self $model) {
            $model->uuid ??= (string) Str::uuid();
        });
    }

    public function isExpired(): bool
    {
        return $this->expires_at->isPast();
    }

    public function hasExceededAttempts(int $max): bool
    {
        return $this->attempts >= $max;
    }

    public function canResend(int $cooldownSeconds): bool
    {
        return $this->last_sent_at->addSeconds($cooldownSeconds)->isPast();
    }

    {

    }

    public function hash(string $code): string
    {

    {

    }

namespace App\Services\Auth;

use App\Enums\Auth\AccountType;
use App\Enums\Auth\RoleName;
use App\Jobs\Auth\SendVerificationCodeJob;
use App\Models\Auth\PendingRegistration;
use App\Models\User;
use App\Repositories\Contracts\Auth\PendingRegistrationRepositoryInterface;
use App\Repositories\Contracts\Auth\UserRepositoryInterface;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Hash;
use Illuminate\Validation\ValidationException;

class RegistrationService
{
    public function __construct(
        private PendingRegistrationRepositoryInterface $pendingRepo,
        private UserRepositoryInterface $userRepo,
        private VerificationCodeService $codeService,
    ) {}

    /**
     * Step 1 — Register: write to pending_registrations. No users row created.
     */
    public function register(string $email, string $plainPassword, ?string $ip): PendingRegistration
    {
        // If a verified user already exists, reject (generic message in controller).
        if ($this->userRepo->findByEmail($email)) {
            throw ValidationException::withMessages([
                'email' => ['An account with this email already exists.'],
            ]);
        }

        $code = $this->codeService->generate();
        $ttl = (int) config('auth.verification_code_ttl_minutes', 15) ?: 15;

        $existing = $this->pendingRepo->findByEmail($email);

        $payload = [
            'email' => $email,
            'password' => Hash::make($plainPassword),
            'verification_code' => $this->codeService->hash($code),
            'attempts' => 0,
            'expires_at' => now()->addMinutes($ttl),
            'last_sent_at' => now(),
            'ip_address' => $ip,
        ];

        $pending = $existing
            ? $this->pendingRepo->update($existing, $payload)
            : $this->pendingRepo->create($payload);

        // Queue the email (Redis queue). Never log the code.
        SendVerificationCodeJob::dispatch($pending->email, $code);

        return $pending;
    }

    /**
     * Step 2 — Verify email: atomically create the users row + assign role.
     */
    public function verifyEmail(string $email, string $code): User
    {
        $pending = $this->pendingRepo->findByEmail($email);

        if (! $pending) {
            throw ValidationException::withMessages([
                'email' => ['No pending registration found for this email.'],
            ]);
        }

        if ($pending->isExpired()) {
            $this->pendingRepo->delete($pending);
            throw ValidationException::withMessages([
                'code' => ['This verification code has expired. Please register again.'],
            ]);
        }

        $maxAttempts = (int) config('auth.verification_max_attempts', 5) ?: 5;

        if ($pending->hasExceededAttempts($maxAttempts)) {
            throw ValidationException::withMessages([
                'code' => ['Too many incorrect attempts. Please register again.'],
            ]);
        }

        if (! $this->codeService->verify($code, $pending->verification_code)) {
            $this->pendingRepo->incrementAttempts($pending);
            throw ValidationException::withMessages([
                'code' => ['The verification code is incorrect.'],
            ]);
        }

        // Atomic: create user + assign role + delete pending row.
        $user = DB::transaction(function () use ($pending) {
            $user = $this->userRepo->create([
                'email' => $pending->email,
                'password' => $pending->password, // already hashed
                'email_verified_at' => now(),
                'account_type' => AccountType::CUSTOMER, // default until onboarding
                'onboarding_completed' => false,
            ]);

            $user->assignRole(RoleName::CUSTOMER->value);

            $this->pendingRepo->delete($pending);

            return $user;
        });

        return $user;
    }

    public function resendVerification(string $email): void
    {
        $pending = $this->pendingRepo->findByEmail($email);

        if (! $pending) {
            // Generic response — do not leak whether email is registered.
            return;
        }

        $cooldown = (int) config('auth.verification_resend_cooldown_seconds', 60) ?: 60;

        if (! $pending->canResend($cooldown)) {
            $wait = $pending->last_sent_at->addSeconds($cooldown)->diffInSeconds(now());
            throw ValidationException::withMessages([
                'email' => ["Please wait {$wait} seconds before requesting a new code."],
            ]);
        }

        $code = $this->codeService->generate();
        $ttl = (int) config('auth.verification_code_ttl_minutes', 15) ?: 15;

        $this->pendingRepo->update($pending, [
            'verification_code' => $this->codeService->hash($code),
            'attempts' => 0,
            'expires_at' => now()->addMinutes($ttl),
            'last_sent_at' => now(),
        ]);

        SendVerificationCodeJob::dispatch($pending->email, $code);
    }
}<?php

class LoginService
{
    public function __construct(
        private UserRepositoryInterface $userRepo,
    ) {}

    public function attempt(string $email, string $password): User
    {
        $user = $this->userRepo->findByEmail($email);

        // Generic message to reduce account enumeration.
        if (! $user || ! $user->hasPassword() || ! Hash::check($password, $user->password)) {
            throw ValidationException::withMessages([
                'email' => ['The provided credentials are incorrect.'],
            ]);
        }

        if (! $user->email_verified_at) {
            throw ValidationException::withMessages([
                'email' => ['Please verify your email address before logging in.'],
            ]);
        }

        return $user;
    }

namespace App\Services\Auth;

use App\Jobs\Auth\SendPasswordResetJob;
use App\Repositories\Contracts\Auth\UserRepositoryInterface;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Str;
use Illuminate\Validation\ValidationException;

class PasswordResetService
{
    public function __construct(
        private UserRepositoryInterface $userRepo,
    ) {}

    public function sendResetLink(string $email): void
    {
        $user = $this->userRepo->findByEmail($email);

        // Always generic; never reveal whether the email exists.
        if (! $user || ! $user->hasPassword()) {
            return;
        }

        $plainToken = Str::random(64);
        $hashedToken = Hash::make($plainToken);
        $ttl = (int) config('auth.password_reset_ttl_minutes', 60) ?: 60;

        DB::table('password_reset_tokens')->updateOrInsert(
            ['email' => $email],
            [
                'token' => $hashedToken,
                'created_at' => now(),
            ],
        );

        // The plain token is emailed to the user; only the hash is stored.
        SendPasswordResetJob::dispatch($email, $plainToken, $ttl);
    }

    public function reset(string $email, string $plainToken, string $newPassword): void
    {
        $user = $this->userRepo->findByEmail($email);

        $record = DB::table('password_reset_tokens')->where('email', $email)->first();

        if (! $user || ! $record || ! Hash::check($plainToken, $record->token)) {
            throw ValidationException::withMessages([
                'token' => ['This password reset token is invalid.'],
            ]);
        }

        $ttl = (int) config('auth.password_reset_ttl_minutes', 60) ?: 60;

        if (now()->diffInMinutes($record->created_at) > $ttl) {
            DB::table('password_reset_tokens')->where('email', $email)->delete();
            throw ValidationException::withMessages([
                'token' => ['This password reset token has expired.'],
            ]);
        }

        DB::transaction(function () use ($user, $email, $newPassword) {
            $this->userRepo->updatePassword($user, Hash::make($newPassword));
            DB::table('password_reset_tokens')->where('email', $email)->delete();
            // Invalidate sessions/tokens on password change.
            $user->tokens()->delete();
        });
    }
}<?php

namespace App\Services\Auth;

use App\Enums\Auth\AccountType;
use App\Enums\Auth\RoleName;
use App\Models\User;
use App\Repositories\Contracts\Auth\SocialAccountRepositoryInterface;
use App\Repositories\Contracts\Auth\UserRepositoryInterface;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Str;
use Laravel\Socialite\Facades\Socialite;

class SocialAuthService
{
    public function __construct(
        private UserRepositoryInterface $userRepo,
        private SocialAccountRepositoryInterface $socialRepo,
    ) {}

    public function redirect(string $provider)
    {
        return Socialite::driver($provider)->redirect();
    }

    public function callback(string $provider): User
    {
        $socialUser = Socialite::driver($provider)->user();

        // 1. Already linked social account?
        $linked = $this->socialRepo->findByProvider($provider, (string) $socialUser->getId());

        if ($linked) {
            return $linked->user;
        }

        // 2. Existing user with same verified email? Link it.
        if ($socialUser->getEmail()) {
            $existing = $this->userRepo->findByEmail($socialUser->getEmail());

            if ($existing) {
                $this->socialRepo->create([
                    'user_id' => $existing->id,
                    'provider' => $provider,
                    'provider_user_id' => (string) $socialUser->getId(),
                ]);

                // Ensure email is marked verified because the provider verified it.
                if (! $existing->email_verified_at) {
                    $this->userRepo->markEmailVerified($existing);
                }

                return $existing->refresh();
            }
        }

        // 3. Brand new user — social email is verified by the provider.
        $user = DB::transaction(function () use ($provider, $socialUser) {
            $email = $socialUser->getEmail() ?? Str::uuid().'@social.local';

            $user = $this->userRepo->create([
                'email' => $email,
                'password' => null,
                'email_verified_at' => now(),
                'account_type' => AccountType::CUSTOMER,
                'onboarding_completed' => false,
            ]);

            $user->assignRole(RoleName::CUSTOMER->value);

            $this->socialRepo->create([
                'user_id' => $user->id,
                'provider' => $provider,
                'provider_user_id' => (string) $socialUser->getId(),
            ]);

            return $user;
        });

        return $user;
    }
}<?php

namespace App\Services\Auth;

use App\Enums\Auth\AccountType;
use App\Models\User;
use App\Repositories\Contracts\Auth\UserRepositoryInterface;
use Illuminate\Validation\ValidationException;

class OnboardingService
{
    public function __construct(
        private UserRepositoryInterface $userRepo,
    ) {}

    public function chooseRole(User $user, string $accountType): User
    {
        if ($user->onboarding_completed) {
            throw ValidationException::withMessages([
                'account_type' => ['Onboarding has already been completed.'],
            ]);
        }

        // Once a user has been approved as an agent, they cannot revert.
        if ($user->hasRole('agent') && $accountType === AccountType::CUSTOMER->value) {
            throw ValidationException::withMessages([
                'account_type' => ['Approved agents cannot switch back to a customer account.'],
            ]);
        }

        $enum = AccountType::tryFrom($accountType);

        if (! $enum || $enum === AccountType::SUPER_ADMIN) {
            throw ValidationException::withMessages([
                'account_type' => ['Invalid account type selected.'],
            ]);
        }

        return $this->userRepo->updateOnboarding($user, [
            'account_type' => $enum,
            'onboarding_completed' => true,
        ]);
    }
}<?php

use App\Http\Controllers\Controller;
use App\Http\Requests\Auth\RegisterRequest;
use App\Http\Requests\Auth\ResendVerificationRequest;
use App\Http\Requests\Auth\VerifyEmailRequest;
use App\Http\Resources\Auth\UserResource;
use App\Services\Auth\RegistrationService;
use Illuminate\Http\JsonResponse;
use Illuminate\Support\Facades\Auth;

class RegistrationController extends Controller
{
    public function __construct(
        private RegistrationService $registrationService,
    ) {}

    public function register(RegisterRequest $request): JsonResponse
    {
        $this->registrationService->register(
            $request->string('email')->toString(),
            $request->string('password')->toString(),
            $request->ip(),
        );

        return response()->json([
            'success' => true,
            'message' => 'A 6-digit verification code has been sent to your email.',
            'data' => null,
        ], 201);
    }

    public function verify(VerifyEmailRequest $request): JsonResponse
    {
        $user = $this->registrationService->verifyEmail(
            $request->string('email')->toString(),
            $request->string('code')->toString(),
        );

        Auth::guard('web')->login($user);
        $request->session()->regenerate();

        return response()->json([
            'success' => true,
            'message' => 'Email verified successfully. Welcome!',
            'data' => new UserResource($user),
        ]);
    }

    public function resend(ResendVerificationRequest $request): JsonResponse
    {
        $this->registrationService->resendVerification(
            $request->string('email')->toString(),
        );

        // Always return the same message to avoid enumeration.
        return response()->json([
            'success' => true,
            'message' => 'If this email is pending verification, a new code has been sent.',
            'data' => null,
        ]);
    }
}<?php

namespace App\Http\Controllers\Auth;

use App\Http\Controllers\Controller;
use App\Http\Requests\Auth\ChangePasswordRequest;
use App\Http\Requests\Auth\ForgotPasswordRequest;
use App\Http\Requests\Auth\ResetPasswordRequest;
use App\Services\Auth\PasswordResetService;
use Illuminate\Http\JsonResponse;
use Illuminate\Support\Facades\Hash;
use Illuminate\Validation\ValidationException;

class PasswordController extends Controller
{
    public function __construct(
        private PasswordResetService $passwordResetService,
    ) {}

    public function forgot(ForgotPasswordRequest $request): JsonResponse
    {
        $this->passwordResetService->sendResetLink(
            $request->string('email')->toString(),
        );

        return response()->json([
            'success' => true,
            'message' => 'If an account exists for this email, a reset link has been sent.',
            'data' => null,
        ]);
    }

    public function reset(ResetPasswordRequest $request): JsonResponse
    {
        $this->passwordResetService->reset(
            $request->string('email')->toString(),
            $request->string('token')->toString(),
            $request->string('password')->toString(),
        );

        return response()->json([
            'success' => true,
            'message' => 'Password reset successfully. Please log in with your new password.',
            'data' => null,
        ]);
    }

    public function change(ChangePasswordRequest $request): JsonResponse
    {
        $user = $request->user();

        if (! $user->hasPassword() || ! Hash::check($request->string('current_password')->toString(), $user->password)) {
            throw ValidationException::withMessages([
                'current_password' => ['The current password is incorrect.'],
            ]);
        }

        $user->forceFill(['password' => Hash::make($request->string('password')->toString())])->save();
        $user->tokens()->delete(); // invalidate any personal access tokens

        return response()->json([
            'success' => true,
            'message' => 'Password changed successfully.',
            'data' => null,
        ]);
    }
}<?php

namespace App\Http\Controllers\Auth;

use App\Http\Controllers\Controller;
use App\Services\Auth\SocialAuthService;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;

class SocialAuthController extends Controller
{
    public function __construct(
        private SocialAuthService $socialAuthService,
    ) {}

    public function redirect(string $provider): RedirectResponse
    {
        $this->ensureSupported($provider);

        return $this->socialAuthService->redirect($provider);
    }

    public function callback(string $provider, Request $request): RedirectResponse
    {
        $this->ensureSupported($provider);

        $user = $this->socialAuthService->callback($provider);

        Auth::guard('web')->login($user);
        $request->session()->regenerate();

        // Redirect to SPA; onboarding is decided on the frontend via /auth/me.
        return redirect()->away(
            rtrim(config('app.frontend_url', env('FRONTEND_URL')), '/').'/auth/social/callback'
        );
    }

    private function ensureSupported(string $provider): void
    {
        abort_unless(in_array($provider, ['google', 'facebook'], true), 404);
    }
}<?php
