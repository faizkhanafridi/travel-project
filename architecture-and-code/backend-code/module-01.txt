    public function create(array $data): User
    {
        $data['uuid'] = $data['uuid'] ?? (string) Str::uuid();

        return User::create($data);
    }

    public function assignRole(User $user, string $role): void
    {
        $user->syncRoles([$role]);
    }

    public function updatePassword(User $user, string $hashedPassword): void
    {
        // use forceFill to bypass the hashed cast (already hashed)
        $user->forceFill(['password' => $hashedPassword])->save();
    }

    public function invalidateSessions(User $user, ?string $exceptSessionId = null): void
    {
        $query = DB::table('sessions')->where('user_id', $user->id);
        if ($exceptSessionId) {
            $query->where('id', '!=', $exceptSessionId);
        }
        $query->delete();
    }

// app/Services/Auth/RegistrationService.php

namespace App\Services\Auth;

use App\Enums\Auth\UserRole;
use App\Mail\Auth\VerificationCodeMail;
use App\Models\Auth\User;
use App\Repositories\Contracts\Auth\PendingRegistrationRepositoryInterface;
use App\Repositories\Contracts\Auth\UserRepositoryInterface;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Facades\Mail;
use Illuminate\Support\Str;
use Symfony\Component\HttpKernel\Exception\UnprocessableEntityHttpException;

class RegistrationService
{
    public function __construct(
        private PendingRegistrationRepositoryInterface $pendingRepo,
        private UserRepositoryInterface $userRepo,
    ) {}

    public function register(string $name, string $email, string $plainPassword, ?string $ip): void
    {
        if ($this->userRepo->findByEmail($email)) {
            throw new UnprocessableEntityHttpException('This email cannot be registered.');
        }

        $this->issueCode($name, $email, $plainPassword, $ip);
    }

    public function resend(string $email): void
    {
        $pending = $this->pendingRepo->findByEmail($email);
        if (! $pending) {
            // generic success to avoid enumeration
            return;
        }

        if ($pending->last_sent_at && $pending->last_sent_at->diffInSeconds(now()) < (int) config('auth.resend_cooldown', 60)) {
            throw new UnprocessableEntityHttpException('Please wait before requesting a new code.');
        }

        $plain = $this->generateCode();
        $this->pendingRepo->upsert($email, [
            'code_hash' => Hash::make($plain),
            'expires_at' => now()->addMinutes((int) config('auth.code_ttl', 10)),
            'last_sent_at' => now(),
            'attempts' => 0,
        ]);

        Mail::to($email)->queue(new VerificationCodeMail($pending->name, $email, $plain));

    }

    public function verify(string $email, string $plainCode, ?string $ip): User
    {
        $pending = $this->pendingRepo->findByEmail($email);
        if (! $pending || $pending->isExpired()) {
            throw new UnprocessableEntityHttpException('Invalid or expired code.');
        }

        if ($pending->attempts >= 5) {
            $this->pendingRepo->delete($pending);
            throw new UnprocessableEntityHttpException('Too many attempts. Please register again.');
        }

        if (! Hash::check($plainCode, $pending->code_hash)) {
            $this->pendingRepo->incrementAttempts($pending);
            throw new UnprocessableEntityHttpException('Invalid or expired code.');
        }

        return DB::transaction(function () use ($pending, $ip) {
            $user = $this->userRepo->create([
                'name' => $pending->name,
                'email' => $pending->email,
                'password' => $pending->password, // already hashed
                'email_verified_at' => now(),
                'status' => \App\Enums\Auth\UserStatus::ACTIVE->value,
                'google_id' => null,
                'facebook_id' => null,
            ]);
            $this->userRepo->assignRole($user, UserRole::CUSTOMER->value);
            $this->pendingRepo->delete($pending);
            return $user;
        });
    }

    private function issueCode(string $name, string $email, string $plainPassword, ?string $ip): void
    {
        $plain = $this->generateCode();
        $this->pendingRepo->upsert($email, [
            'name' => $name,
            'password' => Hash::make($plainPassword),
            'code_hash' => Hash::make($plain),
            'expires_at' => now()->addMinutes((int) config('auth.code_ttl', 10)),
            'last_sent_at' => now(),
            'attempts' => 0,
            'registered_from_ip' => $ip,
        ]);
        Mail::to($email)->queue(new VerificationCodeMail($name, $email, $plain));

    }

    private function generateCode(): string
    {
        return str_pad((string) random_int(0, 999999), 6, '0', STR_PAD_LEFT);
    }

// app/Services/Auth/PasswordResetService.php

namespace App\Services\Auth;

use App\Mail\Auth\PasswordResetCodeMail;
use App\Mail\Auth\SetPasswordCodeMail;
use App\Models\Auth\User;
use App\Repositories\Contracts\Auth\PasswordResetCodeRepositoryInterface;
use App\Repositories\Contracts\Auth\UserRepositoryInterface;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Facades\Mail;
use Symfony\Component\HttpKernel\Exception\UnprocessableEntityHttpException;

class PasswordResetService
{
    public function __construct(
        private PasswordResetCodeRepositoryInterface $codeRepo,
        private UserRepositoryInterface $userRepo,
    ) {}

    public function sendCode(string $email): void
    {
        $user = $this->userRepo->findByEmail($email);
        if (! $user) {
            // generic response to avoid enumeration
            return;
        }

        $existing = $this->codeRepo->findByEmail($email);
        if ($existing && $existing->last_sent_at && $existing->last_sent_at->diffInSeconds(now()) < 60) {
            throw new UnprocessableEntityHttpException('Please wait before requesting a new code.');
        }

        $plain = $this->generateCode();
        $this->codeRepo->upsert($email, [
            'code_hash' => Hash::make($plain),
            'expires_at' => now()->addMinutes((int) config('auth.code_ttl', 10)),
            'last_sent_at' => now(),
            'attempts' => 0,
        ]);

        // Social-only users have no password yet — send a "set password" email
        // instead of a "reset password" email. Both flows reuse the same code table.
        if ($user->password === null) {
            Mail::to($email)->queue(new SetPasswordCodeMail($user->name, $email, $plain));
            return;
        }

        Mail::to($email)->queue(new PasswordResetCodeMail($user->name, $email, $plain));
    }

    public function reset(string $email, string $plainCode, string $newPassword): void
    {
        $user = $this->consumeCode($email, $plainCode);

        if ($user->password === null) {
            // Defensive: reset endpoint must not be used for social-only users.
            // They should use set-password. Still allow it here as a safety net.
            $this->userRepo->updatePassword($user, Hash::make($newPassword));
            $this->userRepo->invalidateSessions($user);
            return;
        }

        $this->userRepo->updatePassword($user, Hash::make($newPassword));
        $this->userRepo->invalidateSessions($user);
    }

    public function setPassword(string $email, string $plainCode, string $newPassword): void
    {
        $user = $this->consumeCode($email, $plainCode);

        if ($user->password !== null) {
            // Defensive: once a password exists, only reset/change-password may be used.
            throw new UnprocessableEntityHttpException('A password is already set for this account.');
        }

        $this->userRepo->updatePassword($user, Hash::make($newPassword));
        $this->userRepo->invalidateSessions($user);
    }

    public function changePassword(User $user, string $currentPassword, string $newPassword): void
    {
        if (! $user->password || ! Hash::check($currentPassword, $user->password)) {
            throw new UnprocessableEntityHttpException('Current password is incorrect.');
        }
        $this->userRepo->updatePassword($user, Hash::make($newPassword));
        // invalidate other sessions
        $this->userRepo->invalidateSessions($user, request()->session()->getId());
    }

    /**
     * Verifies and consumes a password reset / set-password code, returning the user.
     * Shared by reset() and setPassword().
     */
    private function consumeCode(string $email, string $plainCode): User
    {
        $record = $this->codeRepo->findByEmail($email);
        if (! $record || $record->isExpired()) {
            throw new UnprocessableEntityHttpException('Invalid or expired code.');
        }
        if ($record->attempts >= 5) {
            $this->codeRepo->delete($record);
            throw new UnprocessableEntityHttpException('Too many attempts. Please request a new code.');
        }
        if (! Hash::check($plainCode, $record->code_hash)) {
            $this->codeRepo->incrementAttempts($record);
            throw new UnprocessableEntityHttpException('Invalid or expired code.');
        }

        $user = $this->userRepo->findByEmail($email);
        if (! $user) {
            throw new UnprocessableEntityHttpException('Unable to complete the request.');
        }

        $this->codeRepo->delete($record);

        return $user;
    }

    private function generateCode(): string
    {
        return str_pad((string) random_int(0, 999999), 6, '0', STR_PAD_LEFT);
    }

class SocialAuthService
{
    public function __construct(private UserRepositoryInterface $userRepo) {}

    public function redirectUrl(string $provider): string
    {
        return Socialite::driver($provider)
            ->stateless(false)
            ->with(['prompt' => 'select_account'])
            ->redirect()
            ->getTargetUrl();
    }

    public function handleCallback(string $provider): User
    {
        /** @var SocialUser $social */
        $social = Socialite::driver($provider)->stateless(false)->user();

        $email = $social->getEmail();
        if (! $email) {
            throw new \RuntimeException('No email returned from provider.');
        }

        return DB::transaction(function () use ($provider, $social, $email) {
            $existing = User::where('email', $email)->first();
            if ($existing) {
                if ($provider === 'google' && ! $existing->google_id) {
                    $existing->google_id = $social->getId();
                }
                if ($provider === 'facebook' && ! $existing->facebook_id) {
                    $existing->facebook_id = $social->getId();
                }
                $existing->save();
                Auth::login($existing, true);
                request()->session()->regenerate();
                return $existing;
            }

            $user = $this->userRepo->create([
                'name' => $social->getName() ?: Str::before($email, '@'),
                'email' => $email,
                'password' => null,
                'email_verified_at' => now(),
                'status' => UserStatus::ACTIVE->value,
                'google_id' => $provider === 'google' ? $social->getId() : null,
                'facebook_id' => $provider === 'facebook' ? $social->getId() : null,
            ]);
            $this->userRepo->assignRole($user, UserRole::CUSTOMER->value);
            Auth::login($user, true);
            request()->session()->regenerate();
            return $user;
        });
    }

namespace App\Services\Auth;

use App\Models\Auth\User;
use Illuminate\Support\Facades\Auth;
use PragmaRX\Google2FA\Google2FA;
use Symfony\Component\HttpKernel\Exception\UnprocessableEntityHttpException;

class MfaService
{
    public function __construct(private Google2FA $google2fa) {}

    /**
     * @return array{secret: string, otpauth_url: string}
     */
    public function setup(User $user): array
    {
        if ($user->hasMfaEnabled()) {
            throw new UnprocessableEntityHttpException('MFA is already enabled.');
        }

        $secret = $this->google2fa->generateSecretKey();
        $user->forceFill(['two_factor_secret' => $secret])->save();

        $otpauth = $this->google2fa->getQRCodeUrl(
            config('app.name'),
            $user->email,
            $secret
        );

        return ['secret' => $secret, 'otpauth_url' => $otpauth];
    }

    public function confirm(User $user, string $code): void
    {
        if ($user->hasMfaEnabled()) {
            throw new UnprocessableEntityHttpException('MFA is already enabled.');
        }
        if (! $user->two_factor_secret) {
            throw new UnprocessableEntityHttpException('Run MFA setup first.');
        }
        if (! $this->google2fa->verifyKey($user->two_factor_secret, $code)) {
            throw new UnprocessableEntityHttpException('Invalid code.');
        }

        $user->forceFill(['two_factor_confirmed_at' => now()])->save();
    }

    public function verifyChallenge(string $code): User
    {
        $userId = session('2fa:user_id');
        if (! $userId) {
            throw new UnprocessableEntityHttpException('No MFA challenge in progress.');
        }

        /** @var User|null $user */
        $user = User::find($userId);
        if (! $user || ! $user->two_factor_secret || ! $user->hasMfaEnabled()) {
            session()->forget(['2fa:user_id', '2fa:remember']);
            throw new UnprocessableEntityHttpException('Invalid MFA state.');
        }

        if (! $this->google2fa->verifyKey($user->two_factor_secret, $code)) {
            throw new UnprocessableEntityHttpException('Invalid code.');
        }

        $remember = (bool) session('2fa:remember', false);
        session()->forget(['2fa:user_id', '2fa:remember']);

        Auth::login($user, $remember);
        request()->session()->regenerate();

        return $user;
    }

    public function disable(User $user): void
    {
        $user->forceFill([
            'two_factor_secret' => null,
            'two_factor_recovery_codes' => null,
            'two_factor_confirmed_at' => null,
        ])->save();
    }

// app/Services/Auth/AuthService.php

namespace App\Services\Auth;

use App\Enums\Auth\UserStatus;
use App\Models\Auth\User;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\Hash;
use Illuminate\Validation\ValidationException;

class AuthService
{
    /**
     * Attempt to log the user in.
     * @return array{user: User, requires_mfa: bool}
     */
    public function attempt(string $email, string $password, bool $remember = false): array
    {
        /** @var User|null $user */
        $user = User::where('email', $email)->first();

        if (! $user || ! $user->password || ! Hash::check($password, $user->password)) {
            // generic message (no enumeration)
            throw ValidationException::withMessages([
                'email' => ['These credentials do not match our records.'],
            ]);
        }

        if ($user->status !== UserStatus::ACTIVE) {
            throw ValidationException::withMessages([
                'email' => ['This account is not active.'],
            ]);
        }

        if ($user->hasMfaEnabled()) {
            // mark session as MFA-pending; do NOT call Auth::login yet
            session([
                '2fa:user_id' => $user->id,
                '2fa:remember' => $remember,
            ]);
            return ['user' => $user, 'requires_mfa' => true];
        }

        Auth::login($user, $remember);
        request()->session()->regenerate();

        return ['user' => $user, 'requires_mfa' => false];
    }

    public function logout(): void
    {
        Auth::guard('web')->logout();
        request()->session()->invalidate();
        request()->session()->regenerateToken();
    }

        $user = $this->service->verifyChallenge($request->string('code'));
        return (new AuthUserResource($user))->response();
    }

    public function disable(Request $request): JsonResponse
    {
        $this->service->disable($request->user());
        return response()->json(['message' => 'MFA disabled.']);
    }

namespace App\Http\Controllers\Auth;

use App\Http\Controllers\Controller;
use App\Services\Auth\SocialAuthService;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\RedirectResponse;
use Illuminate\Support\Facades\Log;

class SocialAuthController extends Controller
{
    public function __construct(private SocialAuthService $service) {}

    public function redirect(string $provider): JsonResponse
    {
        $this->assertProvider($provider);
        return response()->json(['url' => $this->service->redirectUrl($provider)]);
    }

    public function callback(string $provider): RedirectResponse
    {
        $this->assertProvider($provider);
        try {
            $this->service->handleCallback($provider);
            return redirect()->away(rtrim(config('app.frontend_url'), '/') . '/auth/social/success');
        } catch (\Throwable $e) {
            Log::warning('social login failed', ['provider' => $provider, 'error' => $e->getMessage()]);
            return redirect()->away(rtrim(config('app.frontend_url'), '/') . '/auth/social/error');
        }
    }

    private function assertProvider(string $provider): void
    {
        abort_unless(in_array($provider, ['google', 'facebook'], true), 404);
    }
