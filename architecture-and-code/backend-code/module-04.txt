namespace App\Enums\TravelService;

enum TravelServiceStatus: string
{
    case DRAFT = 'DRAFT';
    case PENDING_REVIEW = 'PENDING_REVIEW';
    case UNDER_REVIEW = 'UNDER_REVIEW';
    case REJECTED = 'REJECTED';
    case PUBLISHED = 'PUBLISHED';
    case UNPUBLISHED = 'UNPUBLISHED';
    case SUSPENDED = 'SUSPENDED';

    public function isEditable(): bool
    {
        return in_array($this, [self::DRAFT, self::REJECTED], true);
    }

    public function isAgentVisible(): bool
    {
        // suspended is hidden from agent UI entirely
        return $this !== self::SUSPENDED;
    }

    public function isPublic(): bool
    {
        return $this === self::PUBLISHED;
    }
}<?php

// app/Repositories/Eloquent/TravelService/TravelServiceRepository.php

namespace App\Repositories\Eloquent\TravelService;

use App\Enums\TravelService\TravelServiceStatus;
use App\Models\Agent\Agent;
use App\Models\TravelService\TravelService;
use App\Repositories\Contracts\TravelService\TravelServiceRepositoryInterface;
use Illuminate\Contracts\Pagination\LengthAwarePaginator;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Str;

class TravelServiceRepository implements TravelServiceRepositoryInterface
{
    public function findByUuid(string $uuid): ?TravelService
    {
        return TravelService::with([
            'category', 'subcategory', 'tags', 'inclusions', 'extras',
            'images', 'requirements', 'history', 'republishRequests', 'agent.user', 'agent.profile',
        ])->where('uuid', $uuid)->first();
    }

    public function findByUuidForAgent(string $uuid, Agent $agent): ?TravelService
    {
        return TravelService::with([
            'category', 'subcategory', 'tags', 'inclusions', 'extras',
            'images', 'requirements', 'history', 'republishRequests',
        ])
        ->where('uuid', $uuid)
        ->where('agent_id', $agent->id)
        ->where('status', '!=', TravelServiceStatus::SUSPENDED->value)
        ->first();
    }

    public function create(array $data): TravelService
    {
        $data['uuid'] = $data['uuid'] ?? (string) Str::uuid();
        return TravelService::create($data);
    }

    public function update(TravelService $service, array $data): TravelService
    {
        $service->fill($data)->save();
        return $service->fresh();
    }

    public function paginateForAgent(Agent $agent, array $filters): LengthAwarePaginator
    {
        $query = TravelService::with(['category', 'subcategory', 'images', 'tags'])
            ->where('agent_id', $agent->id)
            ->where('status', '!=', TravelServiceStatus::SUSPENDED->value);

        if (! empty($filters['status'])) {
            $query->where('status', $filters['status']);
        }
        if (! empty($filters['search'])) {
            $query->where('title', 'like', '%' . $filters['search'] . '%');
        }

        return $query->latest()->paginate((int) ($filters['per_page'] ?? 15));
    }

    public function paginateForAdmin(array $filters): LengthAwarePaginator
    {
        $query = TravelService::with(['agent.user', 'agent.profile', 'category', 'subcategory'])
            ->latest('submitted_at');

        if (! empty($filters['status'])) {
            $query->where('status', $filters['status']);
        }
        if (! empty($filters['search'])) {
            $search = $filters['search'];
            $query->where(function ($q) use ($search) {
                $q->where('title', 'like', "%{$search}%")
                  ->orWhereHas('agent.profile', fn ($p) => $p->where('agency_name', 'like', "%{$search}%"));
            });
        }

        return $query->paginate((int) ($filters['per_page'] ?? 20));
    }

    public function paginatePublic(array $filters): LengthAwarePaginator
    {
        $query = TravelService::with(['agent.profile', 'images', 'tags', 'category', 'subcategory'])
            ->where('status', TravelServiceStatus::PUBLISHED->value);

        if (! empty($filters['subcategory_id'])) {
            $query->where('subcategory_id', $filters['subcategory_id']);
        }
        if (! empty($filters['category_id'])) {
            $query->where('category_id', $filters['category_id']);
        }
        if (! empty($filters['destination_country'])) {
            $query->where('destination_country', $filters['destination_country']);
        }
        if (! empty($filters['destination_city'])) {
            $query->where('destination_city', $filters['destination_city']);
        }
        if (! empty($filters['duration_min'])) {
            $query->where('duration', '>=', (int) $filters['duration_min']);
        }
        if (! empty($filters['duration_max'])) {
            $query->where('duration', '<=', (int) $filters['duration_max']);
        }
        if (! empty($filters['price_min'])) {
            $query->where('price', '>=', (float) $filters['price_min']);
        }
        if (! empty($filters['price_max'])) {
            $query->where('price', '<=', (float) $filters['price_max']);
        }

        $search = $filters['q'] ?? null;
        if ($search) {
            $query->where(function ($q) use ($search) {
                $q->where('title', 'like', "%{$search}%")
                  ->orWhere('short_description', 'like', "%{$search}%")
                  ->orWhere('destination_city', 'like', "%{$search}%")
                  ->orWhere('destination_country', 'like', "%{$search}%")

namespace App\Services\Currency;

use App\Models\Currency\ExchangeRate;
use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\Http;
use Illuminate\Support\Facades\Log;

class CurrencyService
{
    public const SUPPORTED = ['USD', 'EUR', 'GBP', 'PKR', 'AED', 'SAR', 'INR', 'CAD', 'AUD'];

    public function rates(): array
    {
        return Cache::remember('currency:rates:usd', now()->addHours(6), function () {
            $existing = ExchangeRate::where('base', 'USD')->pluck('rate', 'quote')->all();
            if (! empty($existing)) {
                return $existing;
            }
            return $this->refresh();
        });
    }

    public function refresh(): array
    {
        try {
            $response = Http::timeout(6)->get('https://open.er-api.com/v6/latest/USD');
            if (! $response->ok()) {
                throw new \RuntimeException('rates fetch failed');
            }
            $rates = $response->json('rates') ?? [];
            $now = now();
            foreach (self::SUPPORTED as $code) {
                if ($code === 'USD' || ! isset($rates[$code])) {
                    continue;
                }
                ExchangeRate::updateOrCreate(
                    ['base' => 'USD', 'quote' => $code],
                    ['rate' => $rates[$code], 'fetched_at' => $now]
                );
            }
            Cache::forget('currency:rates:usd');
            return ExchangeRate::where('base', 'USD')->pluck('rate', 'quote')->all();
        } catch (\Throwable $e) {
            Log::warning('currency refresh failed', ['error' => $e->getMessage()]);
            return ExchangeRate::where('base', 'USD')->pluck('rate', 'quote')->all();
        }
    }

    public function convert(float $amount, string $from, string $to): float
    {
        $from = strtoupper($from);
        $to = strtoupper($to);
        if ($from === $to) {
            return round($amount, 2);
        }
        $rates = $this->rates();
        $usd = $from === 'USD' ? $amount : ($amount / (float) ($rates[$from] ?? 1));
        return round($usd * (float) ($rates[$to] ?? 1), 2);
    }

namespace App\Services\TravelService;

use App\Enums\TravelService\TravelServiceStatus;
use App\Models\Agent\Agent;
use App\Models\TravelService\ServiceCategory;
use App\Models\TravelService\TravelService;
use App\Repositories\Contracts\TravelService\TravelServiceRepositoryInterface;
use Illuminate\Http\UploadedFile;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Storage;
use Illuminate\Support\Str;
use Symfony\Component\HttpKernel\Exception\AccessDeniedHttpException;
use Symfony\Component\HttpKernel\Exception\UnprocessableEntityHttpException;

class TravelServiceWizardService
{
    public function __construct(private TravelServiceRepositoryInterface $services) {}

    public function create(Agent $agent): TravelService
    {
        return $this->services->create([
            'agent_id' => $agent->id,
            'category_id' => 0,
            'subcategory_id' => 0,
            'title' => 'Untitled draft',
            'slug' => 'draft-' . Str::lower(Str::random(8)),
            'full_description' => '',
            'price' => 0,
            'currency' => 'USD',
            'destination_country' => '',
            'destination_city' => '',
            'duration' => 1,
            'duration_unit' => 'days',
            'status' => TravelServiceStatus::DRAFT->value,
            'attempt' => 0,
        ]);
    }

    public function assertEditable(TravelService $service): void
    {
        if (! $service->status->isEditable()) {
            throw new AccessDeniedHttpException('Service cannot be edited in its current state.');
        }
    }

    public function saveBasic(TravelService $service, array $data): TravelService
    {
        $this->assertEditable($service);

        $sub = ServiceCategory::find($data['subcategory_id']);
        $cat = ServiceCategory::find($data['category_id']);
        if (! $cat || $cat->parent_id !== null) {
            throw new UnprocessableEntityHttpException('Invalid category.');
        }
        if (! $sub || $sub->parent_id !== $cat->id) {
            throw new UnprocessableEntityHttpException('Subcategory does not belong to the selected category.');
        }

        return DB::transaction(function () use ($service, $data) {
            $slug = Str::slug($data['title']) . '-' . Str::lower(Str::random(6));
            $updated = $this->services->update($service, [
                'title' => $data['title'],
                'slug' => $slug,
                'category_id' => $data['category_id'],
                'subcategory_id' => $data['subcategory_id'],
                'step_basic_saved' => true,
            ]);

            $updated->tags()->delete();
            foreach (array_unique($data['tags']) as $tag) {
                $updated->tags()->create(['tag' => $tag]);
            }

            return $updated->fresh();
        });
    }

    public function saveDetails(TravelService $service, array $data): TravelService
    {
        $this->assertEditable($service);
        if (! $service->step_basic_saved) {
            throw new UnprocessableEntityHttpException('Save basic information first.');
        }

        return DB::transaction(function () use ($service, $data) {
            $updated = $this->services->update($service, [
                'short_description' => $data['short_description'] ?? null,
                'full_description' => $data['full_description'],
                'price' => $data['price'],
                'currency' => 'USD',
                'destination_country' => $data['destination_country'],
                'destination_city' => $data['destination_city'],
                'duration' => $data['duration'],
                'duration_unit' => $data['duration_unit'],
                'cancellation_policy' => $data['cancellation_policy'] ?? null,
                'step_details_saved' => true,
            ]);

            $updated->inclusions()->delete();
            foreach (array_values($data['inclusions']) as $i => $label) {
                $updated->inclusions()->create(['label' => $label, 'sort_order' => $i]);
            }

            $updated->extras()->delete();
            foreach ($data['extras'] ?? [] as $i => $extra) {
                $updated->extras()->create([
                    'uuid' => (string) Str::uuid(),
                    'label' => $extra['label'],
                    'price' => $extra['price'],
                    'currency' => 'USD',
                    'description' => $extra['description'] ?? null,
                    'sort_order' => $i,
                ]);
            }

            return $updated->fresh();
        });
    }

    public function uploadGallery(TravelService $service, UploadedFile $cover, array $extraImages): TravelService
    {
        $this->assertEditable($service);
        if (! $service->step_details_saved) {
            throw new UnprocessableEntityHttpException('Save details first.');
        }

        return DB::transaction(function () use ($service, $cover, $extraImages) {
            $path = $this->storeImage($service, $cover);
            $service->images()->create([
                'uuid' => (string) Str::uuid(),
                'file_path' => $path,
                'mime_type' => $cover->getMimeType(),
                'size_bytes' => $cover->getSize(),
                'is_cover' => true,
                'sort_order' => 0,
            ]);

            $i = 1;
            foreach ($extraImages as $img) {
                if (! $img instanceof UploadedFile) continue;
                $p = $this->storeImage($service, $img);
                $service->images()->create([
                    'uuid' => (string) Str::uuid(),
                    'file_path' => $p,
                    'mime_type' => $img->getMimeType(),
                    'size_bytes' => $img->getSize(),
                    'is_cover' => false,
                    'sort_order' => $i++,
                ]);
            }

            return $this->services->update($service, ['step_gallery_saved' => true]);
        });
    }

    public function reorderGallery(TravelService $service, array $orderedUuids, string $coverUuid): TravelService
    {
        $this->assertEditable($service);

        return DB::transaction(function () use ($service, $orderedUuids, $coverUuid) {
            $images = $service->images()->get()->keyBy('uuid');
            $i = 0;
            foreach ($orderedUuids as $uuid) {

            $updated->history()->create([
                'uuid' => (string) Str::uuid(),
                'attempt' => $updated->attempt,
                'action' => 'approved',
                'admin_id' => $admin->id,
            ]);
            return $updated;
        });
    }

        if (! in_array($service->status, [TravelServiceStatus::PENDING_REVIEW, TravelServiceStatus::UNDER_REVIEW], true)) {

        return DB::transaction(function () use ($service, $admin, $reason) {

                'rejection_reason' => $reason,

                'reviewed_by' => $admin->id,

            $updated->history()->create([
                'uuid' => (string) Str::uuid(),
                'action' => 'unpublished',

                       return $updated;

    public function suspend(TravelService $service, User $admin): TravelService
    {
        // Admin cannot suspend if any active (non-completed, non-cancelled) booking exists
        if ($this->services->hasActiveBookings($service)) {
            throw new UnprocessableEntityHttpException('Cannot suspend a service with active bookings.');
        }

        if ($service->status === TravelServiceStatus::SUSPENDED) {
            throw new UnprocessableEntityHttpException('Service is already suspended.');
        }

        return DB::transaction(function () use ($service, $admin) {
            $updated = $this->services->update($service, [
                'status' => TravelServiceStatus::SUSPENDED->value,
                'suspended_at' => now(),
            ]);
            $updated->history()->create([
                'uuid' => (string) Str::uuid(),
                'action' => 'suspended',
                'admin_id' => $admin->id,
            ]);
            return $updated;
        });
    }

    {
        if ($service->status !== TravelServiceStatus::SUSPENDED) {
            throw new UnprocessableEntityHttpException('Service is not suspended.');
        }

        return DB::transaction(function () use ($service, $admin) {
            // Unsuspend restores the service to UNPUBLISHED — admin must republish intentionally
            $updated = $this->services->update($service, [
                'status' => TravelServiceStatus::UNPUBLISHED->value,
                'suspended_at' => null,
            ]);
            $updated->history()->create([
                'uuid' => (string) Str::uuid(),
                'action' => 'unsuspended',
                'admin_id' => $admin->id,
            ]);
            return $updated;
        });

    public function approveRepublish(ServiceRepublishRequest $request, User $admin): TravelService
    {
        if ($request->status !== RepublishRequestStatus::PENDING) {
            throw new UnprocessableEntityHttpException('Request is not pending.');
        }
        $service = $request->service;
        if (! $service || $service->status !== TravelServiceStatus::UNPUBLISHED) {
            throw new UnprocessableEntityHttpException('Service is not in unpublished state.');
        }

        return DB::transaction(function () use ($request, $service, $admin) {
            $request->update([
                'status' => RepublishRequestStatus::APPROVED->value,
                'reviewed_at' => now(),
                'reviewed_by' => $admin->id,
            ]);

            // NO attempt increment. NO new submission record. Just republish the existing service.
            $updated = $this->services->update($service, [
                'status' => TravelServiceStatus::PUBLISHED->value,
                'published_at' => now(),
            ]);

            $updated->history()->create([
                'uuid' => (string) Str::uuid(),
                'action' => 'republish_approved',
                'admin_id' => $admin->id,
            ]);

            return $updated;
        });
    }

    public function rejectRepublish(ServiceRepublishRequest $request, User $admin, string $reason): ServiceRepublishRequest
    {
        if ($request->status !== RepublishRequestStatus::PENDING) {
            throw new UnprocessableEntityHttpException('Request is not pending.');

                'status' => RepublishRequestStatus::REJECTED->value,
                'rejection_reason' => $reason,
                'reviewed_at' => now(),
                'reviewed_by' => $admin->id,
            ]);

            $request->service->history()->create([
                'uuid' => (string) Str::uuid(),
                'action' => 'republish_rejected',
                'reason' => $reason,
                'admin_id' => $admin->id,
            ]);

            return $request->fresh();
        });
    }

class TravelServiceDeletionService
{
    public function __construct(private TravelServiceRepositoryInterface $services) {}

    public function delete(TravelService $service, Agent $agent): void
    {
        if ($service->agent_id !== $agent->id) {
            throw new UnprocessableEntityHttpException('Not your service.');
        }

        // Suspend-aware rule: if suspended, allow deletion only when no bookings and no republish history
        if ($service->status === \App\Enums\TravelService\TravelServiceStatus::SUSPENDED) {
            if ($this->services->hasAnyBookings($service)) {
                throw new UnprocessableEntityHttpException('Cannot delete a suspended service with historical bookings.');
            }
        } else {
            if ($this->services->hasAnyBookings($service)) {
                throw new UnprocessableEntityHttpException('Cannot delete a service with bookings.');
            }
        }

        DB::transaction(function () use ($service) {
            // Delete images from disk
            foreach ($service->images as $image) {
                Storage::disk('private')->delete($image->file_path);
            }

            // Hard delete — soft delete kept available but not used for cleaner data
            $service->delete();
        });
    }
