```
namespace App\Enums\Booking;

enum BookingStatus: string
{

// app/Enums/Booking/BookingOrigin.php

namespace App\Enums\Booking;

enum BookingOrigin: string
{
    case MARKETPLACE = 'marketplace';
    case GLOBAL_POST = 'global_post';
    case SERVICE_REQUEST = 'service_request';

    /** Marketplace + service_request share 2% commission. Global post has 0%. */
    public function commissionRate(): float
    {
        return match ($this) {
            self::MARKETPLACE, self::SERVICE_REQUEST => 0.02,
            self::GLOBAL_POST => 0.0,
        };
    }
}

<?php
```
```php
// app/Repositories/Eloquent/Booking/BookingRepository.php

namespace App\Repositories\Eloquent\Booking;

use App\Enums\Booking\BookingStatus;
use App\Models\Auth\User;
use App\Models\Booking\Booking;
use App\Repositories\Contracts\Booking\BookingRepositoryInterface;
use Illuminate\Contracts\Pagination\LengthAwarePaginator;
use Illuminate\Support\Str;

class BookingRepository implements BookingRepositoryInterface
{
    public function findByUuid(string $uuid): ?Booking
    {
        return Booking::with([
            'customer', 'agent.user', 'agent.profile', 'service', 'post', 'offer',
            'requirementResponses', 'documents', 'milestones', 'escrowEvents',
        ])->where('uuid', $uuid)->first();
    }

    public function findByUuidForUser(string $uuid, User $user): ?Booking
    {
        return Booking::with([
            'customer', 'agent.user', 'agent.profile', 'service', 'post', 'offer',
            'requirementResponses', 'documents', 'milestones', 'escrowEvents',
        ])
        ->where('uuid', $uuid)
        ->where(function ($q) use ($user) {
            $q->where('customer_id', $user->id)
              ->orWhereHas('agent', fn ($a) => $a->where('user_id', $user->id));
        })
        ->first();
    }

    public function create(array $data): Booking
    {
        $data['uuid'] = $data['uuid'] ?? (string) Str::uuid();
        return Booking::create($data);
    }

    public function update(Booking $booking, array $data): Booking
    {
        $booking->fill($data)->save();
        return $booking->fresh();
    }

    public function paginateForCustomer(User $customer, array $filters): LengthAwarePaginator
    {
        $query = Booking::with(['agent.user', 'agent.profile', 'service', 'milestones'])
            ->where('customer_id', $customer->id);

        if (! empty($filters['status'])) {
            $query->where('status', $filters['status']);
        }
        if (! empty($filters['origin'])) {
            $query->where('origin', $filters['origin']);
        }

        return $query->latest()->paginate((int) ($filters['per_page'] ?? 15));
    }

    public function paginateForAgent(int $agentId, array $filters): LengthAwarePaginator
    {
        $query = Booking::with(['customer', 'service', 'milestones'])
            ->where('agent_id', $agentId);

        if (! empty($filters['status'])) {
            $query->where('status', $filters['status']);
        }
        if (! empty($filters['origin'])) {
            $query->where('origin', $filters['origin']);
        }

        return $query->latest()->paginate((int) ($filters['per_page'] ?? 15));
    }

    public function paginateAll(array $filters): LengthAwarePaginator
    {
        $query = Booking::with(['customer', 'agent.user', 'agent.profile', 'service', 'milestones']);

        if (! empty($filters['status'])) {
            $query->where('status', $filters['status']);
        }
        if (! empty($filters['origin'])) {
            $query->where('origin', $filters['origin']);
        }
        if (! empty($filters['search'])) {
            $s = $filters['search'];
            $query->where(function ($q) use ($s) {
                $q->where('title_snapshot', 'like', "%{$s}%")
                  ->orWhereHas('customer', fn ($c) => $c->where('name', 'like', "%{$s}%"))
                  ->orWhereHas('agent.user', fn ($a) => $a->where('name', 'like', "%{$s}%"));
            });
        }

        return $query->latest()->paginate((int) ($filters['per_page'] ?? 20));
    }

    public function findExpiredRefillBookings(): \Illuminate\Support\Collection
    {
        return Booking::where('status', BookingStatus::REFILL_REQUESTED->value)
            ->whereNotNull('auto_cancel_at')
            ->where('auto_cancel_at', '<=', now())
            ->get();
    }
}
```

```php
// app/Services/Booking/BookingCommissionCalculator.php

namespace App\Services\Booking;

use App\Enums\Booking\BookingOrigin;

class BookingCommissionCalculator
{
    /**
     * @return array{commission_rate: float, commission_amount: float, agent_net: float}
     */
    public function calculate(float $total, BookingOrigin $origin): array
    {
        $rate = $origin->commissionRate();
        $commission = round($total * $rate, 2);
        $agentNet = round($total - $commission, 2);

        return [
            'commission_rate' => $rate,
            'commission_amount' => $commission,
            'agent_net' => $agentNet,
        ];
    }

    /**
     * Milestone split — pure math, no commission application (already computed in $agentNet).
     *
     * @return array<int, array{phase: string, percentage: float, amount: float, agent_share_amount: float}>
     */
    public function splitMilestones(float $total, float $agentNet, string $model): array
    {
        if ($model === '1_step') {
            return [[
                'phase' => 'final',
                'percentage' => 100.00,
                'amount' => round($total, 2),
                'agent_share_amount' => round($agentNet, 2),
            ]];
        }

        // 2_step
        return [
            [
                'phase' => 'phase_1',
                'percentage' => 70.00,
                'amount' => round($total * 0.70, 2),
                'agent_share_amount' => round($agentNet * 0.70, 2),
            ],
            [
                'phase' => 'final',
                'percentage' => 30.00,
                'amount' => round($total * 0.30, 2),
                'agent_share_amount' => round($agentNet * 0.30, 2),
            ],
        ];
    }
}
```

```php
// app/Services/Booking/BookingInitiator.php

namespace App\Services\Booking;

use App\Enums\Booking\BookingOrigin;
use App\Enums\Booking\BookingStatus;
use App\Enums\Booking\EscrowEventType;
use App\Enums\Booking\MilestoneModel;

    public function __construct(

        private BookingEscrowRepositoryInterface $escrow,

    /**
     * FLOW 1 — customer books a marketplace service.
     * Note: Finance module is not built — the returned booking sits in REQUESTED
     * and a `pending_finance` escrow event is recorded. When Finance ships, the
     * same method can be extended to capture the payment before switching status.

    public function bookService(User $customer, TravelService $service): Booking

        if ($service->status?->value !== 'PUBLISHED') {
            throw new UnprocessableEntityHttpException('Service is not available.');
        }

            throw new UnprocessableEntityHttpException('You cannot book your own service.');
        }

        return $this->createBooking(
            customer: $customer,
            agent: $service->agent,
            origin: BookingOrigin::MARKETPLACE,
            milestoneModel: $service->category?->milestone_model ?? '1_step',
            service: $service,
            post: null,
            offer: null,
            amountTotal: (float) $service->price,
            title: $service->title,
            duration: (int) $service->duration,
            durationUnit: (string) $service->duration_unit,
            country: (string) $service->destination_country,
            city: (string) $service->destination_city,
        );
    }

    /**
     * FLOW 2 — customer accepts a global-post offer.
     * The offer must already be ACCEPTED via OfferService::decide().
     */
    public function acceptGlobalOffer(User $customer, TravelOffer $offer): Booking
    {
        $post = $offer->post;
        if (! $post || $post->customer_id !== $customer->id) {
            throw new UnprocessableEntityHttpException('Not your post.');
        }
        if ($offer->status?->value !== 'ACCEPTED') {
            throw new UnprocessableEntityHttpException('Offer must be accepted first.');
        }

        return $this->createBooking(
            customer: $customer,
            agent: $offer->agent,
            origin: BookingOrigin::GLOBAL_POST,
            milestoneModel: (string) ($post->milestone_model ?? '1_step'),
            service: null,
            post: $post,
            offer: $offer,
            amountTotal: (float) $offer->amount,
            title: $post->title,
            duration: (int) $post->duration,
            durationUnit: (string) $post->duration_unit,
            country: (string) $post->destination_country,
            city: (string) $post->destination_city,
        );
    }

    /**
     * FLOW 3 — customer accepts an offer attached to a service-request post.
     */
    public function acceptServiceRequestOffer(User $customer, TravelOffer $offer): Booking
    {
        $post = $offer->post;
        if (! $post || $post->customer_id !== $customer->id) {
            throw new UnprocessableEntityHttpException('Not your post.');
        }
        if ($offer->status?->value !== 'ACCEPTED') {
            throw new UnprocessableEntityHttpException('Offer must be accepted first.');
        }

        return $this->createBooking(
            customer: $customer,
            agent: $offer->agent,
            origin: BookingOrigin::SERVICE_REQUEST,
            milestoneModel: (string) ($post->milestone_model ?? '1_step'),
            service: $post->service,
            post: $post,
            offer: $offer,
            amountTotal: (float) $offer->amount,
            title: $post->title,
            duration: (int) $post->duration,
            durationUnit: (string) $post->duration_unit,
            country: (string) $post->destination_country,
            city: (string) $post->destination_city,
        );

    private function createBooking(
        User $customer,
        Agent $agent,
        BookingOrigin $origin,
        string $milestoneModel,
        ?TravelService $service,
        ?TravelPost $post,
        ?TravelOffer $offer,
        float $amountTotal,
        string $title,
        int $duration,
        string $durationUnit,
        string $country,
        string $city,
    ): Booking {
        return DB::transaction(function () use (
            $customer, $agent, $origin, $milestoneModel, $service, $post, $offer,
            $amountTotal, $title, $duration, $durationUnit, $country, $city
        ) {
            $commission = $this->commission->calculate($amountTotal, $origin);

            $booking = $this->bookings->create([
                'customer_id' => $customer->id,
                'agent_id' => $agent->id,
                'travel_service_id' => $service?->id,
                'travel_post_id' => $post?->id,
                'travel_offer_id' => $offer?->id,
                'origin' => $origin->value,
                'milestone_model' => $milestoneModel,
                'status' => BookingStatus::REQUESTED->value,
                'currency' => 'USD',
                'amount_total' => $amountTotal,
                'commission_rate' => $commission['commission_rate'],
                'commission_amount' => $commission['commission_amount'],
                'agent_net' => $commission['agent_net'],
                'title_snapshot' => $title,
                'duration' => $duration,
                'duration_unit' => $durationUnit,
                'destination_country' => $country,
                'destination_city' => $city,
            ]);

            // Milestones are pre-created at 0% released, waiting for the acceptance step
            $splits = $this->commission->splitMilestones($amountTotal, $commission['agent_net'], $milestoneModel);
            foreach ($splits as $split) {
                $this->milestones->create([
                    'booking_id' => $booking->id,
                    'phase' => $split['phase'],
                    'percentage' => $split['percentage'],
                    'amount' => $split['amount'],
                    'agent_share_amount' => $split['agent_share_amount'],
                    'status' => 'PENDING',
                    'submission_round' => 0,
                ]);
            }

            // Escrow placeholder — Finance Coming Soon
            $this->escrow->record(
                $booking->id,
                EscrowEventType::PENDING_FINANCE,
                $amountTotal,
                'USD',
                null,
                [
                    'reason' => 'Module Finance Coming Soon',
                    'origin' => $origin->value,
                    'milestone_model' => $milestoneModel,
                    'next_step' => 'payment_required',
                ],
            );

            return $booking->load(['milestones', 'service', 'post', 'offer']);
        });
    }
```

```php
                $booking->id,
                EscrowEventType::PENDING_FINANCE,
                (float) $booking->amount_total,
                $booking->currency,

                    'action' => 'reverse_charge_on_reject',
                    'next_step' => 'refund_100_percent',

            );

                'status' => BookingStatus::REJECTED->value,
                'cancelled_at' => now(),
                'cancellation_reason' => $reason,
                'refunded_amount' => (float) $booking->amount_total,
            ]);
        });

            'refill_requested_at' => now(),

            'auto_cancel_at' => now()->addDays(7),
```

```php
     */
    public function autoApproveDueMilestones(): int
    {
        $count = 0;
        foreach ($this->milestones->findDueForAutoApproval() as $milestone) {
            try {
                $this->releaseMilestone($milestone->booking, $milestone, 'auto_approved_window_expired', null);
                $count++;
            } catch (\Throwable $e) {
                \Log::warning('auto-approve milestone failed', ['milestone_id' => $milestone->id, 'error' => $e->getMessage()]);
            }
        }
        return $count;
    }

    /**
     * Release the milestone. Called by customer approval or scheduler.
     */
    public function releaseMilestone(Booking $booking, BookingMilestone $milestone, string $reason, ?int $approvedByUserId): Booking
    {
        return DB::transaction(function () use ($booking, $milestone, $reason, $approvedByUserId) {
            $now = now();

            // Phase 1 (2-step) — release 70%; booking stays IN_PROGRESS
            if ($milestone->phase === MilestonePhase::PHASE_1) {
                $this->milestones->update($milestone, [
                    'status' => 'RELEASED',
                    'approved_at' => $now,
                    'released_at' => $now,
                ]);

                // Finance Coming Soon — placeholder escrow event
                $this->escrow->record(
                    $booking->id,
                    EscrowEventType::PENDING_FINANCE,
                    (float) $milestone->agent_share_amount,
                    $booking->currency,
                    $milestone->id,
                    [
                        'reason' => 'Module Finance Coming Soon',
                        'action' => 'release_phase_1',
                        'percentage' => 70,
                        'approved_by' => $approvedByUserId,
                        'trigger' => $reason,
                        'next_step' => 'credit_agent_wallet_70_percent',
                    ],
                );

                $newReleased = (float) $booking->released_amount + (float) $milestone->agent_share_amount;

                return $this->bookings->update($booking, [
                    'released_amount' => $newReleased,
                ]);
            }

            // Final milestone (1-step: entire amount; 2-step: remaining 30%)
            $this->milestones->update($milestone, [
                'status' => 'RELEASED',
                'approved_at' => $now,
                'released_at' => $now,
            ]);

            $this->escrow->record(
                $booking->id,
                EscrowEventType::PENDING_FINANCE,
                (float) $milestone->agent_share_amount,
                $booking->currency,
                $milestone->id,
                [
                    'reason' => 'Module Finance Coming Soon',
                    'action' => 'release_final',
                    'percentage' => (float) $milestone->percentage,
                    'approved_by' => $approvedByUserId,
                    'trigger' => $reason,
                    'next_step' => 'credit_agent_wallet_final',
                ],
            );

            $newReleased = (float) $booking->released_amount + (float) $milestone->agent_share_amount;

            return $this->bookings->update($booking, [
                'released_amount' => $newReleased,
                'status' => BookingStatus::COMPLETED->value,
                'completed_at' => $now,
            ]);
        });
    }
}
```

```php
namespace App\Services\Booking;

use App\Enums\Booking\BookingStatus;
use App\Enums\Booking\EscrowEventType;
use App\Models\Auth\User;
use App\Models\Booking\Booking;

use App\Repositories\Contracts\Booking\BookingRepositoryInterface;

class BookingCancelService
{
    public function __construct(
        private BookingRepositoryInterface $bookings,
        private BookingEscrowRepositoryInterface $escrow,
    ) {}

    public function cancelDueToRefillExpiry(Booking $booking): void
    {
        if ($booking->status !== BookingStatus::REFILL_REQUESTED) {
            return;
        }

        DB::transaction(function () use ($booking) {
            $this->escrow->record(
                $booking->id,
                EscrowEventType::PENDING_FINANCE,
                (float) $booking->amount_total,
                $booking->currency,
                null,
                [
                    'reason' => 'Module Finance Coming Soon',
                    'action' => 'auto_cancel_refill_expired',
                    'next_step' => 'refund_100_percent',
                ],
            );

            $this->bookings->update($booking, [
                'status' => BookingStatus::CANCELLED->value,
                'cancelled_at' => now(),
                'cancellation_reason' => 'Refill requirements not provided within 7 days.',
                'refunded_amount' => (float) $booking->amount_total,
            ]);
        });
    }

    public function customerCancelEarly(User $customer, Booking $booking, string $reason): Booking
    {
        if ($booking->customer_id !== $customer->id) {
            throw new UnprocessableEntityHttpException('Not your booking.');
        }
        if ($booking->status !== BookingStatus::REQUESTED) {
            throw new UnprocessableEntityHttpException('Only REQUESTED bookings may be cancelled by the customer.');
        }

        return DB::transaction(function () use ($booking, $reason) {
            $this->escrow->record(
                $booking->id,
                EscrowEventType::PENDING_FINANCE,
                (float) $booking->amount_total,
                $booking->currency,
                null,
                [
                    'reason' => 'Module Finance Coming Soon',
                    'action' => 'customer_cancel_requested',
                    'next_step' => 'refund_100_percent',
                    'cancel_reason' => $reason,
                ],
            );

            return $this->bookings->update($booking, [
                'status' => BookingStatus::CANCELLED->value,
                'cancelled_at' => now(),
                'cancellation_reason' => $reason,
                'refunded_amount' => (float) $booking->amount_total,
            ]);
        });
    }
}
```

```php
<?php
// app/Services/Booking/BookingExpiryService.php

namespace App\Services\Booking;

use App\Repositories\Contracts\Booking\BookingRepositoryInterface;

class BookingExpiryService
{
    public function __construct(
        private BookingRepositoryInterface $bookings,
        private BookingCancelService $cancel,
        private BookingApprovalService $approval,
    ) {}

    /**
     * Runs from the scheduler. Handles:
     *  - 7-day refill guard auto-cancel
     *  - Milestone window auto-approval (72h / 168h)
     *
     * @return array{cancelled: int, auto_approved: int}
     */
    public function sweep(): array
    {
        $cancelled = 0;
        foreach ($this->bookings->findExpiredRefillBookings() as $booking) {
            $this->cancel->cancelDueToRefillExpiry($booking);
            $cancelled++;
        }

        $autoApproved = $this->approval->autoApproveDueMilestones();

        return ['cancelled' => $cancelled, 'auto_approved' => $autoApproved];
    }
}
```

### `app/Http/Controllers/Booking/BookingController.php`

```php
<?php
// app/Http/Controllers/Booking/BookingController.php

namespace App\Http\Controllers\Booking;

use App\Http\Controllers\Controller;
use App\Http\Requests\Booking\AgentDecisionRequest;
use App\Http\Requests\Booking\RefillRequirementsRequest;
use App\Http\Requests\Booking\SubmitRequirementsRequest;
use App\Http\Resources\Booking\BookingDetailResource;
use App\Http\Resources\Booking\BookingListResource;
use App\Models\Agent\Agent;
use App\Models\Booking\Booking;
use App\Models\TravelPost\TravelOffer;
use App\Models\TravelService\TravelService;
use App\Repositories\Contracts\Booking\BookingRepositoryInterface;
use App\Services\Booking\BookingCancelService;
use App\Services\Booking\BookingDecisionService;
use App\Services\Booking\BookingInitiator;
use App\Services\Booking\BookingRequirementsService;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Symfony\Component\HttpKernel\Exception\AccessDeniedHttpException;
use Symfony\Component\HttpKernel\Exception\UnprocessableEntityHttpException;

class BookingController extends Controller
{
    public function __construct(
        private BookingRepositoryInterface $bookings,
        private BookingInitiator $initiator,
        private BookingRequirementsService $requirements,
        private BookingDecisionService $decisions,
        private BookingCancelService $cancel,
    ) {}

    // ---------------- Customer endpoints ----------------

    public function indexCustomer(Request $request): JsonResponse
    {
        $paginator = $this->bookings->paginateForCustomer($request->user(), $request->only(['status', 'origin', 'per_page']));
        return BookingListResource::collection($paginator)->response();
    }

    public function indexAgent(Request $request): JsonResponse
    {
        $agent = Agent::where('user_id', $request->user()->id)->first();
        abort_unless($agent, 403, 'Only agents have bookings.');
        $paginator = $this->bookings->paginateForAgent($agent->id, $request->only(['status', 'origin', 'per_page']));
        return BookingListResource::collection($paginator)->response();
    }

    public function show(Request $request, string $uuid): JsonResponse
    {
        $booking = $this->bookings->findByUuidForUser($uuid, $request->user());
        abort_unless($booking, 404, 'Booking not found.');
        return (new BookingDetailResource($booking))->response();
    }

    // Flow 1 — book a marketplace service
    public function bookService(Request $request, string $serviceUuid): JsonResponse
    {
        $service = TravelService::with(['agent', 'category'])->where('uuid', $serviceUuid)->first();
        abort_unless($service, 404, 'Service not found.');

        $booking = $this->initiator->bookService($request->user(), $service);

        return (new BookingDetailResource($booking->load(['customer', 'agent.user', 'agent.profile', 'service', 'milestones', 'documents', 'requirementResponses', 'escrowEvents'])))
            ->additional([
                'finance' => [
                    'status' => 'coming_soon',
                    'message' => 'Module Finance Coming Soon',
                    'next_step' => 'submit_requirements',
                ],
            ])
            ->response()
            ->setStatusCode(201);
    }

    // Flow 2 & 3 — accept an offer (global post or service-request)
    public function acceptOffer(Request $request, string $offerUuid): JsonResponse
    {
        $offer = TravelOffer::with(['post.service', 'agent'])->where('uuid', $offerUuid)->first();
        abort_unless($offer, 404, 'Offer not found.');

        $post = $offer->post;
        $booking = $post->type?->value === 'service_request'
            ? $this->initiator->acceptServiceRequestOffer($request->user(), $offer)
            : $this->initiator->acceptGlobalOffer($request->user(), $offer);

        return (new BookingDetailResource($booking->load(['customer', 'agent.user', 'agent.profile', 'post', 'offer', 'service', 'milestones', 'documents', 'requirementResponses', 'escrowEvents'])))
            ->additional([
                'finance' => [
                    'status' => 'coming_soon',
                    'message' => 'Module Finance Coming Soon',
                    'next_step' => 'submit_requirements',
                ],
            ])
            ->response()
            ->setStatusCode(201);
    }

    // ---------------- Customer lifecycle endpoints ----------------

    public function submitRequirements(SubmitRequirementsRequest $request, string $uuid): JsonResponse
    {
        $booking = $this->bookings->findByUuidForUser($uuid, $request->user());
        abort_unless($booking, 404, 'Booking not found.');

        $booking = $this->requirements->submit(
            $request->user(),
            $booking,
            $request->input('responses', []),
            $request->filesByKey(),
        );

        return (new BookingDetailResource($booking->load(['customer', 'agent.user', 'agent.profile', 'service', 'post', 'offer', 'milestones', 'documents', 'requirementResponses', 'escrowEvents'])))->response();
    }

    public function customerCancel(Request $request, string $uuid): JsonResponse
    {
        $request->validate(['reason' => ['required', 'string', 'min:5', 'max:2000']]);
        $booking = $this->bookings->findByUuidForUser($uuid, $request->user());
        abort_unless($booking, 404, 'Booking not found.');

        $booking = $this->cancel->customerCancelEarly($request->user(), $booking, (string) $request->string('reason'));

        return (new BookingDetailResource($booking->load(['milestones', 'escrowEvents'])))->response();
    }

    // ---------------- Agent lifecycle endpoints ----------------

    public function agentDecision(AgentDecisionRequest $request, string $uuid): JsonResponse
    {
        $booking = $this->bookings->findByUuid($uuid);
        abort_unless($booking, 404, 'Booking not found.');

        $decision = (string) $request->string('decision');
        $booking = $decision === 'accept'
            ? $this->decisions->accept($request->user(), $booking)
            : $this->decisions->reject($request->user(), $booking, (string) $request->string('reason'));

        return (new BookingDetailResource($booking->load(['customer', 'agent.user', 'agent.profile', 'service', 'post', 'offer', 'milestones', 'documents', 'requirementResponses', 'escrowEvents'])))->response();
    }

    public function agentRequestRefill(RefillRequirementsRequest $request, string $uuid): JsonResponse
    {
        $booking = $this->bookings->findByUuid($uuid);
        abort_unless($booking, 404, 'Booking not found.');

        $booking = $this->decisions->requestRefill($request->user(), $booking, (string) $request->string('reason'));

        return (new BookingDetailResource($booking->load(['milestones', 'requirementResponses', 'escrowEvents'])))->response();
    }

    // ---------------- Admin ----------------

    public function adminIndex(Request $request): JsonResponse
    {
        abort_unless($request->user()->hasRole('super_admin'), 403);
        $paginator = $this->bookings->paginateAll($request->only(['status', 'origin', 'search', 'per_page']));
        return BookingListResource::collection($paginator)->response();
    }
}
```

### `app/Http/Controllers/Booking/BookingMilestoneController.php`

```php
<?php
// app/Http/Controllers/Booking/BookingMilestoneController.php

namespace App\Http\Controllers\Booking;

use App\Http\Controllers\Controller;
use App\Http\Requests\Booking\ApproveMilestoneRequest;
use App\Http\Requests\Booking\UploadDocumentsRequest;
use App\Http\Resources\Booking\BookingDetailResource;
use App\Repositories\Contracts\Booking\BookingRepositoryInterface;
use App\Services\Booking\BookingApprovalService;
use App\Services\Booking\BookingDocumentService;
use App\Services\Booking\BookingMilestoneService;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;

class BookingMilestoneController extends Controller
{
    public function __construct(
        private BookingRepositoryInterface $bookings,
        private BookingDocumentService $documents,
        private BookingMilestoneService $milestones,
        private BookingApprovalService $approval,
    ) {}

    public function uploadDocuments(UploadDocumentsRequest $request, string $uuid): JsonResponse
    {
        $booking = $this->bookings->findByUuid($uuid);
        abort_unless($booking, 404, 'Booking not found.');

        $phase = (string) $request->string('phase');
        foreach ($request->input('documents', []) as $i => $doc) {
            $file = $request->file("documents.{$i}.file");
            if (! $file) continue;
            $this->documents->upload(
                $request->user(),
                $booking,
                $phase,
                (string) $doc['label'],
                $file,
            );
        }

        $booking = $this->bookings->findByUuid($uuid);

        return (new BookingDetailResource($booking->load(['milestones', 'documents', 'escrowEvents'])))->response();
    }

    public function submitMilestone(Request $request, string $uuid, string $phase): JsonResponse
    {
        $booking = $this->bookings->findByUuid($uuid);
        abort_unless($booking, 404, 'Booking not found.');

        $milestone = $booking->milestones()->where('phase', $phase)->first();
        abort_unless($milestone, 404, 'Milestone not found.');

        $milestone = $this->milestones->submitMilestone($request->user(), $booking, $milestone);
        $booking = $this->bookings->findByUuid($uuid);

        return (new BookingDetailResource($booking->load(['milestones', 'documents', 'escrowEvents'])))
            ->additional(['next_step' => 'customer_verification_window_started'])
            ->response();
    }

    public function approve(ApproveMilestoneRequest $request, string $uuid, string $phase): JsonResponse
    {
        $booking = $this->bookings->findByUuid($uuid);
        abort_unless($booking, 404, 'Booking not found.');

        $milestone = $booking->milestones()->where('phase', $phase)->first();
        abort_unless($milestone, 404, 'Milestone not found.');

        $action = (string) $request->string('action');

        if ($action === 'approve') {
            $booking = $this->approval->approve($request->user(), $booking, $milestone);
            $next = 'milestone_released_finance_coming_soon';
        } else {
            $this->approval->requestResubmit($request->user(), $booking, $milestone, (string) $request->string('reason'));
            $booking = $this->bookings->findByUuid($uuid);
            $next = 'agent_must_resubmit';
        }

        return (new BookingDetailResource($booking->load(['milestones', 'documents', 'escrowEvents'])))
            ->additional([
                'finance' => [
                    'status' => 'coming_soon',
                    'message' => 'Module Finance Coming Soon',
                ],
                'next_step' => $next,
            ])
            ->response();
    }
}
```

```php
<?php
// app/Http/Controllers/Booking/BookingDocumentController.php

namespace App\Http\Controllers\Booking;

use App\Http\Controllers\Controller;
use App\Models\Booking\BookingDocument;
use App\Models\Booking\BookingRequirementResponse;
use App\Services\Booking\BookingDocumentService;
use Illuminate\Support\Facades\Storage;
use Symfony\Component\HttpFoundation\StreamedResponse;

class BookingDocumentController extends Controller
{
    public function __construct(private BookingDocumentService $documents) {}

    public function download(BookingDocument $document): StreamedResponse
    {
        return $this->documents->stream(request()->user(), $document);
    }

    public function requirementFile(string $uuid): StreamedResponse
    {
        $response = BookingRequirementResponse::where('uuid', $uuid)->first();
        abort_unless($response && $response->file_path, 404, 'File not found.');

        $booking = $response->booking;
        $user = request()->user();
        $allowed = $user && (
            $booking->customer_id === $user->id
            || $booking->agent?->user_id === $user->id
            || $user->hasRole('super_admin')
        );
        abort_unless($allowed, 403, 'Not authorized.');

        $disk = Storage::disk('private');
        abort_unless($disk->exists($response->file_path), 404, 'File not found.');

        return $disk->response($response->file_path, $response->file_original_name, [
            'Content-Type' => $response->file_mime_type,
            'Cache-Control' => 'private, no-store',
        ]);
    }
}
```
