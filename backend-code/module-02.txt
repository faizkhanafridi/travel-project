namespace App\Services\UserProfile;

use App\Models\User;
use App\Models\UserProfile\UserProfile;
use App\Repositories\Contracts\UserProfile\LanguageRepositoryInterface;
use App\Repositories\Contracts\UserProfile\UserProfileRepositoryInterface;
use Illuminate\Http\UploadedFile;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Storage;
use Illuminate\Validation\ValidationException;

class UserProfileService
{
    public function __construct(
        private UserProfileRepositoryInterface $profileRepo,
        private LanguageRepositoryInterface $languageRepo,
    ) {}

    public function forUser(User $user): ?UserProfile
    {
        return $this->profileRepo->findByUser($user);
    }

    public function findByUsername(string $username): ?UserProfile
    {
        return $this->profileRepo->findByUsername($username);
    }

    public function createOrUpdate(User $user, array $data, array $languageIds = [], ?UploadedFile $photo = null): UserProfile
    {
        // Username uniqueness check (against other profiles).
        $existing = $this->profileRepo->findByUser($user);
        if ($this->profileRepo->usernameExists($data['username'], $existing?->id)) {
            throw ValidationException::withMessages([
                'username' => ['This username is already taken.'],
            ]);
        }

        // Validate languages exist.
        $validIds = $this->languageRepo->existingIds($languageIds);
        if (count($validIds) !== count($languageIds)) {
            throw ValidationException::withMessages([
                'language_ids' => ['One or more selected languages are invalid.'],
            ]);
        }

        return DB::transaction(function () use ($user, $data, $validIds, $photo, $existing) {
            if ($photo) {
                $path = $this->storePhoto($photo);

                // Delete previous photo after successful write.
                if ($existing?->profile_photo_path) {
                    Storage::disk(config('filesystems.profile_photo_disk', 'public'))
                        ->delete($existing->profile_photo_path);
                }

                $data['profile_photo_path'] = $path;
            }

            $profile = $existing
                ? $this->profileRepo->update($existing, $data)
                : $this->profileRepo->create($user, $data);

            $this->profileRepo->syncLanguages($profile, $validIds);

            return $profile->fresh('languages');
        });
    }

    public function updatePhoto(User $user, UploadedFile $photo): UserProfile
    {
        $profile = $this->profileRepo->findByUser($user);

        if (! $profile) {
            throw ValidationException::withMessages([
                'photo' => ['Create your profile before uploading a photo.'],
            ]);
        }

        return DB::transaction(function () use ($profile, $photo) {
            $path = $this->storePhoto($photo);

            if ($profile->profile_photo_path) {
                Storage::disk(config('filesystems.profile_photo_disk', 'public'))
                    ->delete($profile->profile_photo_path);
            }

            return $this->profileRepo->update($profile, ['profile_photo_path' => $path]);
        });
    }

    public function removePhoto(User $user): UserProfile
    {
        $profile = $this->profileRepo->findByUser($user);

        if (! $profile) {
            throw ValidationException::withMessages([
                'photo' => ['No profile found.'],
            ]);
        }

        if ($profile->profile_photo_path) {
            Storage::disk(config('filesystems.profile_photo_disk', 'public'))
                ->delete($profile->profile_photo_path);
        }

        return $this->profileRepo->update($profile, ['profile_photo_path' => null]);
    }

    private function storePhoto(UploadedFile $photo): string
    {
        $disk = config('filesystems.profile_photo_disk', 'public');
        $path = config('filesystems.profile_photo_path', 'user-profiles/photos');

        // store() generates a random filename — never trust the client filename.
        return $photo->store($path, $disk);
    }
}<?php

namespace App\Services\UserProfile;

use App\Enums\UserProfile\EmailChangeMethod;
use App\Enums\UserProfile\EmailChangeStep;
use App\Jobs\UserProfile\SendEmailChangeCodeJob;
use App\Models\User;
use App\Models\UserProfile\EmailChangeRequest;
use App\Repositories\Contracts\Auth\UserRepositoryInterface;
use App\Repositories\Contracts\UserProfile\EmailChangeRequestRepositoryInterface;
use App\Services\Auth\VerificationCodeService;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Hash;
use Illuminate\Validation\ValidationException;

class EmailChangeService
{
    public function __construct(
        private EmailChangeRequestRepositoryInterface $repo,
        private UserRepositoryInterface $userRepo,
        private VerificationCodeService $codeService,
    ) {}

    /**
     * Standard flow step 1: user submits new email → code goes to CURRENT email.
     */
    public function initiateWithCurrentEmail(User $user, string $newEmail): EmailChangeRequest
    {
        $this->assertEmailAvailable($user, $newEmail);

        $code = $this->codeService->generate();
        $ttl = (int) config('user_profile.email_change_code_ttl', 15);

        $this->invalidateActive($user);

        $request = $this->repo->create($user, [
            'new_email' => $newEmail,
            'method' => EmailChangeMethod::CURRENT_EMAIL_CODE->value,
            'step' => EmailChangeStep::AWAITING_CURRENT_EMAIL_CODE->value,
            'current_email_code' => $this->codeService->hash($code),
            'current_email_attempts' => 0,
            'current_email_expires_at' => now()->addMinutes($ttl),
            'last_sent_at' => now(),
        ]);

        SendEmailChangeCodeJob::dispatch($user->email, $code, 'current');

        return $request;
    }

    /**
     * Verify the code sent to the CURRENT email; then send code to NEW email.
     */
    public function verifyCurrentEmailCode(User $user, string $code): EmailChangeRequest
    {
        $request = $this->requireRequest($user, EmailChangeStep::AWAITING_CURRENT_EMAIL_CODE);

        $this->assertNotExpired($request->current_email_expires_at);
        $this->assertAttempts($request->current_email_attempts);

        if (! $this->codeService->verify($code, (string) $request->current_email_code)) {
            $this->repo->update($request, [
                'current_email_attempts' => $request->current_email_attempts + 1,
            ]);
            throw ValidationException::withMessages([
                'code' => ['The verification code is incorrect.'],
            ]);
        }

        // Issue code for NEW email.
        $newCode = $this->codeService->generate();
        $ttl = (int) config('user_profile.email_change_code_ttl', 15);

        $this->repo->update($request, [
            'step' => EmailChangeStep::AWAITING_NEW_EMAIL_CODE->value,
            'new_email_code' => $this->codeService->hash($newCode),
            'new_email_attempts' => 0,
            'new_email_expires_at' => now()->addMinutes($ttl),
            'last_sent_at' => now(),
        ]);
        $request->refresh();

        SendEmailChangeCodeJob::dispatch(
            $request->new_email,
            $newCode,
            'new'
        );

        return $request;
    }

    /**
     * Alternative flow: user provides password → code sent to NEW email.
     */
    public function initiateWithPassword(User $user, string $newEmail, string $password): EmailChangeRequest
    {
        $this->assertEmailAvailable($user, $newEmail);

        if (! $user->hasPassword() || ! Hash::check($password, $user->password)) {
            throw ValidationException::withMessages([
                'current_password' => ['The current password is incorrect.'],
            ]);
        }

        $code = $this->codeService->generate();
        $ttl = (int) config('user_profile.email_change_code_ttl', 15);

        $this->invalidateActive($user);

        $request = $this->repo->create($user, [
            'new_email' => $newEmail,
            'method' => EmailChangeMethod::PASSWORD->value,
            'step' => EmailChangeStep::PASSWORD_VERIFIED_AWAITING_NEW_EMAIL_CODE->value,
            'new_email_code' => $this->codeService->hash($code),
            'new_email_attempts' => 0,
            'new_email_expires_at' => now()->addMinutes($ttl),
            'last_sent_at' => now(),
        ]);

        SendEmailChangeCodeJob::dispatch($newEmail, $code, 'new');

        return $request;
    }

    /**
     * Complete: verify code sent to new email and finally update the user's email.
     */
    public function verifyNewEmailCode(User $user, string $code): User
    {
        $request = $this->requireRequest(
            $user,
            EmailChangeStep::AWAITING_NEW_EMAIL_CODE,
            EmailChangeStep::PASSWORD_VERIFIED_AWAITING_NEW_EMAIL_CODE,
        );

        $this->assertNotExpired($request->new_email_expires_at);
        $this->assertAttempts($request->new_email_attempts);

        if (! $this->codeService->verify($code, (string) $request->new_email_code)) {
            $this->repo->update($request, [
                'new_email_attempts' => $request->new_email_attempts + 1,
            ]);
            throw ValidationException::withMessages([
                'code' => ['The verification code is incorrect.'],
            ]);
        }

        $this->assertEmailAvailable($user, $request->new_email);

        return DB::transaction(function () use ($user, $request) {
            $user->forceFill([
                'email' => $request->new_email,
                'email_verified_at' => now(),
            ])->save();

            $this->repo->delete($request);

            return $user->refresh();
        });
    }

    private function requireRequest(User $user, EmailChangeStep ...$allowedSteps): EmailChangeRequest
    {
        $request = $this->repo->activeForUser($user);

        if (! $request) {
            throw ValidationException::withMessages([
                'email' => ['No active email change request.'],
            ]);
        }

        $allowed = array_map(fn($s) => $s->value, $allowedSteps);
        if (! in_array($request->step, $allowed, true)) {
            throw ValidationException::withMessages([
                'email' => ['This step is not valid for the current request.'],
            ]);
        }

        return $request;
    }

    private function invalidateActive(User $user): void
    {
        $existing = $this->repo->activeForUser($user);
        if ($existing) {
            $this->repo->delete($existing);
        }
    }

    private function assertEmailAvailable(User $user, string $newEmail): void
    {
        if (strtolower($user->email) === strtolower($newEmail)) {
            throw ValidationException::withMessages([
                'new_email' => ['The new email is the same as your current email.'],
            ]);
        }

        if ($this->userRepo->findByEmail($newEmail)) {
            throw ValidationException::withMessages([
                'new_email' => ['This email is already in use.'],
            ]);
        }
    }

    private function assertNotExpired($expiresAt): void
    {
        if (! $expiresAt || $expiresAt->isPast()) {
            throw ValidationException::withMessages([
                'code' => ['This verification code has expired. Please start again.'],
            ]);
        }
    }

    private function assertAttempts(int $attempts): void
    {
        $max = (int) config('user_profile.email_change_max_attempts', 5);
        if ($attempts >= $max) {
            throw ValidationException::withMessages([
                'code' => ['Too many incorrect attempts. Please start again.'],
            ]);
        }
    }
}

namespace App\Http\Controllers\UserProfile;

use App\Http\Controllers\Controller;
use App\Http\Requests\UserProfile\CreateUserProfileRequest;
use App\Http\Requests\UserProfile\UpdatePhotoRequest;
use App\Http\Requests\UserProfile\UpdateUserProfileRequest;
use App\Http\Resources\UserProfile\UserProfileResource;
use App\Services\UserProfile\UserProfileService;
use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;
use Illuminate\Validation\ValidationException;

class UserProfileController extends Controller
{
    public function __construct(
        private UserProfileService $profileService,
    ) {}

    // GET /api/profile — returns the authenticated user's profile (or null).
    public function show(Request $request): JsonResponse
    {
        $profile = $this->profileService->forUser($request->user());

        return response()->json([
            'success' => true,
            'message' => $profile ? 'Profile retrieved.' : 'No profile created yet.',
            'data' => $profile ? new UserProfileResource($profile->load('languages', 'user')) : null,
        ]);
    }

    // POST /api/profile — create profile (idempotent: will create or update).
    public function store(CreateUserProfileRequest $request): JsonResponse
    {
        $profile = $this->profileService->createOrUpdate(
            $request->user(),
            $request->safe()->except(['language_ids', 'photo']),
            $request->input('language_ids', []),
            $request->file('photo'),
        );

        return response()->json([
            'success' => true,
            'message' => 'Profile saved.',
            'data' => new UserProfileResource($profile->load('languages', 'user')),
        ]);
    }

    // PUT /api/profile — update existing profile.
    public function update(UpdateUserProfileRequest $request): JsonResponse
    {
        if (! $this->profileService->forUser($request->user())) {
            throw ValidationException::withMessages([
                'profile' => ['No profile exists. Create one first.'],
            ]);
        }

        $profile = $this->profileService->createOrUpdate(
            $request->user(),
            $request->safe()->except(['language_ids', 'photo']),
            $request->input('language_ids', []),
            $request->file('photo'),
        );

        return response()->json([
            'success' => true,
            'message' => 'Profile updated.',
            'data' => new UserProfileResource($profile->load('languages', 'user')),
        ]);
    }

    // POST /api/profile/photo — replace photo.
    public function updatePhoto(UpdatePhotoRequest $request): JsonResponse
    {
        $profile = $this->profileService->updatePhoto($request->user(), $request->file('photo'));

        return response()->json([
            'success' => true,
            'message' => 'Photo updated.',
            'data' => new UserProfileResource($profile->load('languages', 'user')),
        ]);
    }

    // DELETE /api/profile/photo — remove photo.
    public function destroyPhoto(Request $request): JsonResponse
    {
        $profile = $this->profileService->removePhoto($request->user());

        return response()->json([
            'success' => true,
            'message' => 'Photo removed.',
            'data' => new UserProfileResource($profile->load('languages', 'user')),
        ]);
    }
}<?php

class PublicProfileController extends Controller
{
    public function __construct(
        private UserProfileService $profileService,
    ) {}

    // GET /api/u/{username} — public customer profile. NEVER returns private fields.
    public function show(string $username): JsonResponse
    {
        $profile = $this->profileService->findByUsername($username);

        abort_unless($profile, 404, 'Profile not found.');

        // Agent public profiles belong to Module 03 — not exposed here.
        if ($profile->user?->hasRole('agent')) {
            abort(404, 'Profile not found.');
        }

        return response()->json([
            'success' => true,
            'message' => 'Public profile retrieved.',
            'data' => new PublicUserProfileResource($profile->load('languages', 'user')),
        ]);
    }
}<?php

use App\Http\Requests\UserProfile\VerifyEmailChangeCodeRequest;
use App\Http\Resources\Auth\UserResource;
use App\Services\UserProfile\EmailChangeService;
use Illuminate\Http\JsonResponse;

class EmailChangeController extends Controller
{
    public function __construct(
        private EmailChangeService $emailChangeService,
    ) {}

    // POST /api/profile/email/initiate — code goes to CURRENT email.
    public function initiate(InitiateEmailChangeRequest $request): JsonResponse
    {
        $this->emailChangeService->initiateWithCurrentEmail(
            $request->user(),
            $request->string('new_email')->toString(),
        );

        return response()->json([
            'success' => true,
            'message' => 'A 6-digit code has been sent to your current email address.',
            'data' => null,
        ]);
    }

    // POST /api/profile/email/initiate-with-password — alternative flow.
    public function initiateWithPassword(InitiateEmailChangeWithPasswordRequest $request): JsonResponse
    {
        $this->emailChangeService->initiateWithPassword(
            $request->user(),
            $request->string('new_email')->toString(),
            $request->string('current_password')->toString(),
        );

        return response()->json([
            'success' => true,
            'message' => 'A 6-digit code has been sent to your new email address.',
            'data' => null,
        ]);
    }

    // POST /api/profile/email/verify-current — verifies code sent to current email.
    public function verifyCurrent(VerifyEmailChangeCodeRequest $request): JsonResponse
    {
        $this->emailChangeService->verifyCurrentEmailCode(
            $request->user(),
            $request->string('code')->toString(),
        );

        return response()->json([
            'success' => true,
            'message' => 'Current email verified. A code has been sent to your new email address.',
            'data' => null,
        ]);
    }

    // POST /api/profile/email/verify-new — verifies code sent to new email; finalizes change.
    public function verifyNew(VerifyEmailChangeCodeRequest $request): JsonResponse
    {
        $user = $this->emailChangeService->verifyNewEmailCode(
            $request->user(),
            $request->string('code')->toString(),
        );

        return response()->json([
            'success' => true,
            'message' => 'Email address updated successfully.',
            'data' => new UserResource($user),
        ]);
    }
}<?php
